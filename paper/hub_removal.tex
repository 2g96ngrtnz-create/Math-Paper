\title[Hub removal in randomly oriented divisor graphs]{Hub removal in randomly oriented divisor graphs:\\ the giant strong component is the smooth fibre}
\author{[Author]}
\date{Draft, September 2026}

\begin{document}

\begin{abstract}
Let $\cD_\rho(N)$ be the randomly oriented divisor graph of Kim and Phillips: the divisor graph on $\{1,\dots,N\}$, with each edge oriented from multiple to divisor and then independently reversed with probability $\rho$. Let $\Phi_K$ be the size of the largest strongly connected component after the vertices $1,\dots,K$ are deleted. We prove that for every fixed $\rho\in(0,1)$,
\[
0\;\le\;\Psi\Bigl(N,\frac{N}{K+1}\Bigr)-\E[\Phi_K]\;\le\;\frac{30\,N}{\log\log N}\qquad(0\le K<N,\ N\ge N_0(\rho)),
\]
where $\Psi(x,y)$ is the number of $y$-smooth integers up to $x$. Consequently, if $K=N^{\theta+o(1)}$ with $0\le\theta\le 1$ (that is, $\log(K+1)/\log N\to\theta$), then $\E[\Phi_K]/N\to\dick\bigl(1/(1-\theta)\bigr)$, where $\dick$ is Dickman's function and $\dick(\infty)=0$. For $0\le\theta<1$ the same limit holds if the $K$ vertices of largest degree are deleted instead, whether degrees are computed once or recomputed after each deletion. This answers Question~20 of Kim and Phillips. Moreover, with high probability the largest strong component consists of all but $o(N)+O(K)$ of the $N/(K+1)$-smooth vertices, and the second largest strong component has expected size at most $K+o(N)$.

The upper bound comes from the decomposition of the divisor graph into fibres of the rough part. The lower bound rests on one observation. Every divisibility edge $\{x,x'\}$ with $x'\le N^{1-2\delta}$ is shadowed by at least $N^{2\delta}-2$ independent directed $2$-paths through common multiples. So with high probability the strong components are at least as coarse as the connected components of a deterministic divisor graph on $(K,N^{1-2\delta}]$, and a doubling argument shows that the smooth part of that graph is connected. The proof is elementary and self-contained. It uses the estimates of Chebyshev and Mertens, a second-moment bound for the number of small prime factors, and, to identify the limit, Dickman's asymptotic for $\Psi$; all three are proved here from first principles. Every lemma, proposition, theorem and corollary has been formally verified in Lean~4 with Mathlib, and the proofs given here follow the formal ones.
\end{abstract}

\subsection{The model and the question}
For $N\ge 2$ the \emph{divisor graph} $G_N$ has vertex set $[N]=\{1,\dots,N\}$ and an edge $\{a,b\}$ whenever $a\ne b$ and $a\mid b$ or $b\mid a$. Kim and Phillips~\cite{KP} introduced the \emph{randomly oriented divisor graph} $\cD_\rho(N)$. Each edge $\{d,m\}$ with $d\mid m$, $d<m$, is first oriented $m\to d$ (multiple to divisor) and then independently reversed with probability $\rho\in[0,1]$. Thus every edge independently points $m\to d$ with probability $1-\rho$ and $d\to m$ with probability $\rho$. Write $\#\Phi(D)$ for the number of vertices of a largest strongly connected component (SCC) of a directed graph $D$. The main theorem of~\cite{KP} states that $\E[\#\Phi(\cD_\rho(N))]/N\to1$ for every fixed $\rho\in(0,1)$. The authors ask:

\begin{question}[{\cite[Question~20]{KP}}]\label{q:KP}
What can be said about the expected size of the largest strongly connected component as we remove the vertices of highest degrees, one by one. In other words, how will the expected size of the largest strongly connected component change as we remove the vertices labeled $1,2,3,4$, and so on?
\end{question}

For $0\le K<N$ let $V_{N,K}=\{K+1,\dots,N\}$, let $G_{N,K}=G_N[V_{N,K}]$ be the induced subgraph, and let
\[
\cD_\rho(N,K):=\cD_\rho(N)[V_{N,K}]
\]
be the random directed graph obtained from $\cD_\rho(N)$ by deleting the vertices $1,\dots,K$. Its edges are exactly the edges of $G_{N,K}$, oriented independently as above. Put
\[
\Phi_K:=\#\Phi\bigl(\cD_\rho(N,K)\bigr).
\]
Throughout, $\rho\in(0,1)$ is the reversal probability, and we write
\[
q:=\rho(1-\rho)\in(0,\tfrac14],\qquad \lambda:=\max(\rho,1-\rho)\in[\tfrac12,1).
\]

\subsection{Notation}
$p$ always denotes a prime. $\Pp(n)$ is the largest prime factor of $n$, with $\Pp(1)=1$. For real $x\ge0$ and $y\ge1$,
\[
\Psi(x,y):=\#\{1\le n\le x:\ \Pp(n)\le y\}.
\]
For $z\ge2$, $\omega_z(n):=\#\{p\le z:\ p\mid n\}$. We write $\dick$ for Dickman's function (\S\ref{sec:prelim}); the letter $\rho$ alone always means the reversal probability. For $0\le K<N$ put
\[
B=B(N,K):=\Bigl\lfloor\frac{N}{K+1}\Bigr\rfloor,\qquad F_1=F_1(N,K):=\{m\in V_{N,K}:\ \Pp(m)\le B\}.
\]
Note that $\Pp(m)\le N/(K+1)$ if and only if $\Pp(m)\le B$, so $\Psi(N,N/(K+1))=\Psi(N,B)$. We call $F_1$ the \emph{smooth fibre}. Logarithms are natural. ``With high probability'' (w.h.p.) means with probability $1-o(1)$ as $N\to\infty$.

\begin{theorem}[Main theorem]\label{thm:main}\status{PROVEN}\lean{thm_main}
Fix $\rho\in(0,1)$. For all $N\ge2$ and $0\le K<N$ we have $\Phi_K\le \Psi(N,N/(K+1))$ deterministically, and
\[
\lim_{N\to\infty}\ \max_{0\le K<N}\ \frac1N\Bigl(\Psi\Bigl(N,\frac N{K+1}\Bigr)-\E[\Phi_K]\Bigr)=0 .
\]
\end{theorem}

\begin{theorem}[Rate]\label{thm:rate}\status{PROVEN}\lean{thm_rate}
For every $\rho\in(0,1)$ there is $N_0(\rho)$ such that for all $N\ge N_0(\rho)$ and all $0\le K<N$,
\[
0\le \Psi\Bigl(N,\frac N{K+1}\Bigr)-\E[\Phi_K]\le \frac{30\,N}{\log\log N}.
\]
\end{theorem}

The bound is non-trivial only when $\log\log N>30$. Its content is the uniformity in $K$ and the explicit rate, not the numerical constant, which we have not tried to optimise.

For $K\le\sqrt N-1$ this becomes fully explicit (Remark~\ref{rem:smallK}):
\begin{equation}\label{eq:smallK}
1-\frac{\E[\Phi_K]}{N}=\log\frac{\log N}{\log B}+O_\rho\Bigl(\frac1{\log\log N}\Bigr),\qquad B=\Bigl\lfloor\frac N{K+1}\Bigr\rfloor .
\end{equation}
So, up to an additive error $O_\rho(1/\log\log N)$ that is uniform in $0\le K\le\sqrt N-1$, the expected fraction of vertices lost after removing the hubs $1,\dots,K$ equals $\log\frac{\log N}{\log B}$. This main term dominates the error only when $\log K\gg\log N/\log\log N$. For smaller $K$, in particular for bounded $K$, we do not determine the loss to within a factor $1+o(1)$.

\begin{corollary}[Limit profile]\label{cor:profile}\status{PROVEN}\lean{cor_profile, dickLim_continuous, dickLim_strictAntiOn, dickLim_of_le_half}
Fix $\rho\in(0,1)$, and let $K=K(N)$ satisfy $0\le K(N)<N$ for all large $N$ and $\log(K+1)/\log N\to\theta$. (Necessarily $\theta\in[0,1]$.) Then
\[
\lim_{N\to\infty}\frac{\E[\Phi_K]}{N}=\dick\Bigl(\frac1{1-\theta}\Bigr),
\]
with the convention $\dick(\infty)=0$. The profile $\theta\mapsto\dick(1/(1-\theta))$, extended by $0$ for $\theta\ge1$, is continuous on $\mathbb R$ and strictly decreasing on $[0,1]$. It equals $1-\log\frac1{1-\theta}$ for $0\le\theta\le\frac12$.
\end{corollary}

\begin{corollary}[Structure of the giant]\label{cor:structure}\status{PROVEN}\lean{cor_structure, cor_structure_prob}
Fix $\rho\in(0,1)$. Let $\Phi_K^{*}$ be a largest SCC of $\cD_\rho(N,K)$ (chosen by any rule), and let $\Phi_K^{(2)}$ be the size of a largest SCC other than $\Phi_K^*$ ($0$ if there is none). For $N\ge N_0(\rho)$ and all $0\le K<N$:
\begin{enumerate}[label=\textup{(\alph*)}]
\item $\E\bigl|\Phi_K-\Psi(N,\frac N{K+1})\bigr|\le 30N/\log\log N$. In particular $\Phi_K/N-\Psi(N,\frac N{K+1})/N\to0$ in probability, uniformly in $K$.
\item $\E\,|F_1\,\triangle\,\Phi^*_K|\le 2K+30N/\log\log N$.
\item $\E[\Phi^{(2)}_K]\le K+30N/\log\log N$.
\end{enumerate}
\end{corollary}

So when $K=o(N)$, the following holds for every fixed $\varepsilon>0$ with probability $1-o(1)$, by Markov's inequality applied to (b) and (c). The largest SCC differs from the set of $N/(K+1)$-smooth integers in $(K,N]$ in at most $\varepsilon N$ vertices, and every other SCC has at most $\varepsilon N$ vertices. The term $K$ in (c) cannot be dropped in general; see Remark~\ref{rem:second}.

\begin{corollary}[Removing the vertices of highest degree]\label{cor:degree}\status{PROVEN}\lean{cor_degree_static, cor_degree_adaptive, static_eq_Icc}
Fix $\rho\in(0,1)$ and let $\log(K+1)/\log N\to\theta\in[0,1)$. Let $T=T(N)\subset[N]$ with $|T|=K$ be chosen from the \emph{undirected} graph $G_N$ alone (and so independently of the orientation), for all large $N$, in either of the following ways:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{static attack}: $\min_{t\in T}\deg_{G_N}(t)\ge\max_{v\notin T}\deg_{G_N}(v)$;
\item \emph{adaptive attack}: $T=\{u_1,\dots,u_K\}$, where for each $j$ the vertex $u_{j+1}$ has maximal degree in $G_N-\{u_1,\dots,u_j\}$.
\end{enumerate}
Ties may be broken arbitrarily. Then $\E[\#\Phi(\cD_\rho(N)-T)]/N\to\dick(1/(1-\theta))$. Moreover, for fixed $\kappa\in(0,\frac12)$ and $N\ge N_1(\kappa)$, if $1\le K\le N^{1/2-\kappa}$ then the static set $T$ is exactly $\{1,\dots,K\}$.
\end{corollary}

Since the total degree of a vertex in any orientation equals its degree in $G_N$, (i) and (ii) cover attacks by total degree in the directed graph. Attacks by in- or out-degree depend on the orientation and are not covered; see \S\ref{sec:open}.

\subsection{Discussion}
\emph{Answer to Question~\ref{q:KP}.} Deleting any $K=N^{o(1)}$ hubs costs nothing asymptotically; for instance $\E[\Phi_K]/N\to1$ for every fixed $K$. Deleting $K=N^{\theta}$ hubs, a vanishing fraction of the vertices, leaves a giant of relative size $\dick(1/(1-\theta))$. For example, deleting the $\sqrt N$ hubs leaves $1-\log2\approx30.7\%$. Deleting $N^{1-o(1)}$ hubs destroys the giant. The vertices $1,\dots,K$ carry a $\theta$-fraction of the $N\log N+O(N)$ edges (Remark~\ref{rem:edges}), so for moderate $\theta$ hub removal is far more damaging than the loss of edges alone suggests: at $\theta=\frac12$ about half of the edges disappear, but only $30.7\%$ of the vertices remain in the giant. This is the fragility to targeted attacks familiar from scale-free networks~\cite{AJB,BR}. In the divisor graph, however, the surviving giant has an explicit arithmetic description: the smooth fibre. At finite $N$ the convergence is slow (compare Theorem~\ref{thm:rate}). For instance, for $K=1$ every prime $p\in(N/3,N/2]$ lies in $F_1$ but has degree~$1$ in $G_{N,1}$ (its only neighbour is $2p$), so it is never in a strong component of size $\ge2$. Hence $\Psi(N,B)-\E[\Phi_1]\ge\pi(N/2)-\pi(N/3)$, which is $\sim N/(6\log N)$ by the prime number theorem~\cite{Had,dlVP}. This is a loss that vanishes asymptotically but is visible at moderate $N$ (see (O1) in \S\ref{sec:open}).

\emph{What is new.} To our knowledge Theorems~\ref{thm:main} and~\ref{thm:rate} and Corollaries~\ref{cor:profile}--\ref{cor:degree} are new. The fibre decomposition behind the upper bound (Lemma~\ref{lem:fibres}) is a simple and essentially known observation; a closely related statement appears in McNew's work on the divisor graph~\cite{McN}. The inputs from analytic number theory (Mertens, Chebyshev, Dickman, a Tur\'an-type variance bound, and the divisor bound) are classical. The lower-bound mechanism (Lemmas~\ref{lem:robust} and~\ref{lem:doubling}) is, as far as we know, new. It differs from the triangle argument of~\cite{KP}, which uses paths through the vertex~$1$, and it does not need a strongly connected ``hub clique''. Such a clique, for instance the interval $(K,N^{1/2-\delta}]$, whose elements pairwise have many common multiples, can capture only the integers with a divisor in the clique. For $\theta>1/3$ that should miss a positive proportion of $F_1$; see the sketch in Remark~\ref{rem:clique}.

\subsection{Idea of the proof}
\emph{Upper bound.} If $d\mid m$ with $K<d<m\le N$, then $m/d\le B$. Hence the $B$-rough part of a vertex is constant on connected components, and every SCC lies in a fibre $\{v:\ v=n s,\ \Pp(s)\le B\}$, which has at most $\Psi(N,B)$ elements.

\emph{Lower bound.} Fix a small $\delta>0$ and put $M=N^{1-2\delta}$.
\begin{enumerate}[label=(\arabic*)]
\item \emph{Robust edges.} If $K<x<x'\le M$ and $x\mid x'$, then each of the $\ge N^{2\delta}-2$ multiples $y$ of $x'$ in $(x',N]$ gives a directed path $x\to y\to x'$ with probability $q$, independently over $y$; likewise for $x'\to y\to x$. A union bound shows that w.h.p.\ every edge of the \emph{deterministic} graph $G_{M,K}$ joins two vertices of the same SCC. So every connected component of $G_{M,K}$ lies inside one SCC.
\item \emph{Doubling.} All $x\in(K,M]$ with $\Pp(x)\le M/(2(K+1))$ lie in one component of $G_{M,K}$. Multiply by $2$ until just below $M$, divide by an odd prime, and repeat; this ends at the power of $2$ in $(M/2,M]$. Call this set $S_M$.
\item \emph{Attachment.} A vertex $m\in(M,N]$ joins the SCC containing $S_M$ as soon as, among the edges from $m$ to its divisors in $S_M$, at least one points each way. This has probability $1-\rho^{s}-(1-\rho)^s$, where $s$ is the number of such divisors. An elementary counting argument shows that all but $o(N)$ elements of $F_1$ have $s\ge s_0$ for any fixed $s_0$. It uses a reserve of $s_0$ small prime factors of $m$ (supplied by Tur\'an's variance bound), a deterministic case analysis, and Mertens' theorem.
\end{enumerate}

\section{Arithmetic preliminaries}\label{sec:prelim}
%======================================================================

We use the following classical facts. All of them are proved in this paper, with the explicit constants given by the formal proofs. The longer proofs of Lemmas~\ref{lem:mertens} and~\ref{lem:dickman} are in Appendix~\ref{app:classical}. For real arguments, $\Psi(x,y)=\Psi(\lfloor x\rfloor,\lfloor y\rfloor)$, and a sum over $p\le y$ is a sum over $p\le\lfloor y\rfloor$.

\begin{lemma}[Mertens~\cite{Mertens}, Chebyshev; see {\cite[Ch.~XXII]{HW}}, {\cite[Part~I, Ch.~1]{Ten}}]\label{lem:mertens}\status{PROVEN}\lean{lemma_2_1, mertens_a, mertens_a_real, mertens_b, mertens_b_nat, mertens_c, mertens_d}
Put $c_0:=\log4+2$ and $C_1:=18$. Then:
\begin{enumerate}[label=\textup{(\alph*)}]
\item $\sum_{p\le y}\frac{\log p}{p-1}\le\log y+c_0$ for every integer $y\ge1$ and every real $y\ge2$;
\item $\bigl|\sum_{y<p\le w}\frac1p-\log\frac{\log w}{\log y}\bigr|\le\frac{C_1}{\log y}$ for all real $2\le y\le w$; for integers $2\le y\le w$ the bound $8/\log y$ holds;
\item $\sum_{p\le z}\frac1p\ge\log\log z-13\ge\log\log z-C_1$ for all real $z\ge2$;
\item $\pi(x)\le 5x/\log x\le C_1x/\log x$ for all real $x\ge2$.
\end{enumerate}
\end{lemma}
\begin{proof}
See Appendix~\ref{app:classical}. The proof uses only Chebyshev's bound $\prod_{p\le n}p\le4^n$, Legendre's formula for $v_p(n!)$, the inequalities $n^n/e^n\le n!\le n^n$, and Abel summation.
\end{proof}

\begin{lemma}[Dickman's function~\cite{Dickman,dB,Hil}; see {\cite[Part~III, Ch.~5]{Ten}}, \cite{Gra,HT}]\label{lem:dickman}\status{PROVEN}\lean{dickman, dickman_unique, dickman_integral, mul_dickman_eq_integral, dickman_hasDerivAt, dickman_pos, dickman_le_one, dickman_eq_one_sub_log, dickman_strictAntiOn, dickman_asymptotic, dickman_tendstoUniformlyOn}
There is a unique function $\dick:[0,\infty)\to\mathbb R$ that is continuous, equals $1$ on $[0,1]$, and is differentiable at every $u>1$ with $u\,\dick'(u)=-\dick(u-1)$. For $u\ge1$ it satisfies
\[
\dick(u)=1-\int_1^u\frac{\dick(w-1)}{w}\,dw\qquad\text{and}\qquad u\,\dick(u)=\int_{u-1}^u\dick(v)\,dv .
\]
Moreover $0<\dick\le1$, $\dick(u)=1-\log u$ for $1\le u\le2$, and $\dick$ is strictly decreasing on $[1,\infty)$. For every fixed $U\ge1$,
\[
\sup_{1\le u\le U}\Bigl|\frac{\Psi(x,x^{1/u})}{x}-\dick(u)\Bigr|\longrightarrow0\qquad(x\to\infty).
\]
\end{lemma}
\begin{proof}
See Appendix~\ref{app:classical}. The function is constructed recursively on the intervals $[n,n+1]$. The asymptotic is proved by induction on $\lceil U\rceil$ from Buchstab's identity (Lemma~\ref{lem:buchstab}), Lemma~\ref{lem:mertens}(b),(d) and a Riemann-sum estimate for sums over primes (Lemma~\ref{lem:primesum}). No result from the literature on smooth numbers is used. The pointwise asymptotic is due to Dickman~\cite{Dickman}; de Bruijn~\cite{dB} and Hildebrand~\cite{Hil} give error terms.
\end{proof}

\begin{lemma}[Elementary smooth-number bound]\label{lem:psi-upper}\status{PROVEN, standard}\lean{psi_upper, psi_upper', sum_log_smooth_le, dickman_le_two_div, dickman_tendsto_zero}
For integers $x\ge2$ and $y\ge1$, with $c_0=\log4+2$,
\[
\Psi(x,y)\le\sqrt x+\frac{2x(\log y+c_0)}{\log x}.
\]
In particular $\dick(u)\le 2/u$ for all $u\ge1$, and $\dick(u)\to0$ as $u\to\infty$.
\end{lemma}
\begin{proof}
For $n\ge1$ we have $\log n=\sum_{p^k\mid n,\ k\ge1}\log p$. Hence
\begin{align*}
\sum_{\substack{n\le x\\ \Pp(n)\le y}}\log n&=\sum_{\substack{p\le y\\ k\ge1}}\log p\cdot\#\{n\le x:\ \Pp(n)\le y,\ p^k\mid n\}\\
&\le x\sum_{p\le y}\log p\sum_{k\ge1}p^{-k}=x\sum_{p\le y}\frac{\log p}{p-1}\le x(\log y+c_0)
\end{align*}
by Lemma~\ref{lem:mertens}(a) for the integer $y\ge1$. On the other hand, at least $\Psi(x,y)-\sqrt x$ of the $n$ counted satisfy $n>\sqrt x$, and each of them has $\log n>\frac12\log x$. Therefore $(\Psi(x,y)-\sqrt x)\cdot\frac12\log x\le x(\log y+c_0)$.

For the last assertion, fix $u\ge1$. For an integer $N\ge2$ put $y:=\lfloor N^{1/u}\rfloor\ge1$, so $\Psi(N,N^{1/u})=\Psi(N,y)$ and $\log y\le\frac1u\log N$. The bound gives
\[
\frac{\Psi(N,N^{1/u})}N\le N^{-1/2}+\frac2u+\frac{2c_0}{\log N}.
\]
As $N\to\infty$ the left side tends to $\dick(u)$ by Lemma~\ref{lem:dickman}, and the right side tends to $2/u$. So $\dick(u)\le2/u$, and $0<\dick(u)\le2/u$ gives $\dick(u)\to0$.
\end{proof}
\uses{Lemma~\ref{lem:mertens}(a); for the consequence, Lemma~\ref{lem:dickman}.}

\begin{lemma}[Variance of $\omega_z$; the special case of Tur\'an--Kubilius we need]\label{lem:turan}\status{PROVEN, standard}\lean{turan_variance, turan_count}
Let $N\ge1$ and $2\le z\le N$ be integers, and $L:=\sum_{p\le z}\frac1p$. Then
\[
\sum_{m\le N}\bigl(\omega_z(m)-L\bigr)^2\le 3NL .
\]
(The variance bound needs only $z\le N$. For real $z$ nothing changes, since $\omega_z=\omega_{\lfloor z\rfloor}$ and $L$ depends only on $\lfloor z\rfloor$.) Consequently, for every real $s\le L/2$,
\[
\#\{m\le N:\ \omega_z(m)<s\}\le\frac{12N}{L}.
\]
\end{lemma}
\begin{proof}
Write $f=\omega_z$. We have $S_1:=\sum_{m\le N}f(m)=\sum_{p\le z}\lfloor N/p\rfloor\ge NL-\pi(z)$. Also
\[
S_2:=\sum_{m\le N}f(m)^2=\sum_{p\le z}\lfloor N/p\rfloor+\sum_{\substack{p,p'\le z\\ p\ne p'}}\lfloor N/(pp')\rfloor\le NL+NL^2,
\]
because $m$ is divisible by the distinct primes $p,p'$ if and only if $pp'\mid m$. Hence
\[
\sum_{m\le N}(f(m)-L)^2=S_2-2LS_1+NL^2\le NL+NL^2-2L(NL-\pi(z))+NL^2=NL+2L\pi(z)\le 3NL,
\]
using $\pi(z)\le z\le N$. If $f(m)<s\le L/2$, then $|f(m)-L|>L/2$. By Chebyshev's inequality the number of such $m$ is at most $3NL/(L/2)^2=12N/L$.
\end{proof}
\uses{nothing beyond counting.}

\begin{remark}[The Tur\'an--Kubilius inequality]\label{rem:TK}\status{PROVEN}\lean{remark_2_5, turan_kubilius, tk_core, tk_E, tk_AG}
Lemma~\ref{lem:turan} is the special case $f=\omega_z$ of the Tur\'an--Kubilius inequality~\cite{Turan}; see \cite[Part~III, Ch.~3]{Ten}. That inequality states the following. There is an absolute constant $C$ (one may take $C=130$) such that for every complex additive function $f$ (that is, $f(mn)=f(m)+f(n)$ whenever $(m,n)=1$) and every $x\ge2$,
\[
\sum_{n\le x}|f(n)-A(x)|^2\le C\,x\,B(x)^2,
\]
where $A(x)=\sum_{p^\nu\le x}f(p^\nu)p^{-\nu}(1-p^{-1})$ and $B(x)^2=\sum_{p^\nu\le x}|f(p^\nu)|^2p^{-\nu}$. We use only the self-contained special case proved above.

\emph{Proof with $C=130$.} Put $N=\lfloor x\rfloor$. For a prime power $q=p^\nu\le N$ ($\nu\ge1$) put $w_q=q^{-1}(1-p^{-1})$ and $X_q(n)=\mathbf 1[v_p(n)=\nu]$, so that $f(n)=\sum_{q\le N}f(q)X_q(n)$ for $1\le n\le N$. Splitting $f$ into real and imaginary parts, and each of these into positive and negative parts, costs a factor $2$. It reduces the claim to $\sum_{n\le N}(F(n)-A)^2\le65NB^2$, where $F=\sum_qg_qX_q$ with weights $g_q\ge0$, $A=\sum_qg_qw_q$ and $B^2=\sum_qg_q^2/q$. Now $\#\{n\le N:v_p(n)=\nu\}\ge Nw_q-1$. For $q=p^\nu$ and $q'=r^\mu$ with $p\ne r$, inclusion--exclusion gives $\#\{n\le N:v_p(n)=\nu,\ v_r(n)=\mu\}\le Nw_qw_{q'}+2\cdot\mathbf 1[qq'\le N]$; for $p=r$ and $\nu\ne\mu$ this count is $0$. Expanding the square therefore gives
\[
\sum_{n\le N}(F(n)-A)^2\le NB^2+2E+2AG,\qquad E=\sum_{qq'\le N}g_qg_{q'},\quad G=\sum_{q\le N}g_q .
\]
The prime powers $p^\nu\le Y$ with $\nu\ge2$ have $p\le\sqrt Y$ and $\nu\le\log_2Y$. With Lemma~\ref{lem:mertens}(d) this shows that there are at most $58Y/\log 2Y$ prime powers $q\le Y$. Also $\sum_{q\le M}1/q\le1+\log M$. By Cauchy--Schwarz, $E^2\le B^4\sum_{qq'\le N}qq'\le(22NB^2)^2$, where each pair is split according to which of $q,q'$ is at most $\sqrt N$; and $(AG)^2\le B^2(1+\log N)\cdot B^2\sum_{q\le N}q\le(10NB^2)^2$. Hence the sum is at most $(1+44+20)NB^2$.\qed
\end{remark}

\begin{lemma}[Divisor bound {\cite[Thm.~315]{HW}}]\label{lem:divisor}\status{PROVEN, standard}\lean{divisor_bound}
For every $\varepsilon>0$ we have $\tau(n)\le C_\varepsilon n^\varepsilon$ for all $n\ge1$, where $C_\varepsilon:=\max\bigl(1,\frac1{\varepsilon\log2}\bigr)^{\lceil 2^{1/\varepsilon}\rceil}$.
\end{lemma}
\begin{proof}
Put $c:=\max(1,1/(\varepsilon\log2))$ and $B:=2^{1/\varepsilon}$. We have $\tau(n)=\prod_{p^a\,\|\,n}(a+1)$ and $n^\varepsilon=\prod_{p^a\,\|\,n}p^{a\varepsilon}$. If $p\ge B$, then $p^{a\varepsilon}\ge2^a\ge a+1$. If $p<B$, then $2^{a\varepsilon}=e^{a\varepsilon\log2}\ge1+a\varepsilon\log2$, so $c\,p^{a\varepsilon}\ge c\,2^{a\varepsilon}\ge c+a\,c\,\varepsilon\log2\ge1+a$. Hence $\tau(n)\le c^{\#\{p\mid n:\ p<B\}}\,n^\varepsilon\le c^{\lceil B\rceil}n^\varepsilon$, since at most $\lceil B\rceil$ primes are $<B$ and $c\ge1$.
\end{proof}

\begin{lemma}[Limit of the smooth count]\label{lem:psi-limit}\status{PROVEN, standard}\lean{psi_limit, psi_eq_PsiR_rpow, psi_div_le}
Let $K=K(N)$ satisfy $0\le K(N)<N$ for all large $N$ and $\log(K+1)/\log N\to\theta$. (Necessarily $\theta\in[0,1]$.) Then
\[
\lim_{N\to\infty}\frac{\Psi(N,N/(K+1))}{N}=\dick\Bigl(\frac1{1-\theta}\Bigr),\qquad\text{with }\dick(\infty)=0.
\]
\end{lemma}
\begin{proof}
Put $r_N:=\log(K+1)/\log N$ and $y:=N/(K+1)$. For large $N$ we have $N\ge2$ and $K<N$, so $0\le r_N\le1$ and $B=\lfloor y\rfloor\ge1$.

If $\theta<1$, then $r_N<1$ for large $N$. Put $u_N:=1/(1-r_N)\ge1$. Then $N^{1/u_N}=N^{1-r_N}=\exp(\log N-\log(K+1))=y$ exactly, so $\Psi(N,N/(K+1))=\Psi(N,N^{1/u_N})$. Also $u_N\to u:=1/(1-\theta)$, so $u_N\in[1,u+1]$ for large $N$. Lemma~\ref{lem:dickman} with $U=u+1$ gives $\Psi(N,N^{1/u_N})/N-\dick(u_N)\to0$, and $\dick(u_N)\to\dick(u)$ by continuity of $\dick$.

If $\theta\ge1$, then $\theta=1$, since $r_N\le1$. As $B\ge1$ and $\log B\le\log N-\log(K+1)$, Lemma~\ref{lem:psi-upper} gives
\[
\frac{\Psi(N,B)}N\le N^{-1/2}+2(1-r_N)+\frac{2c_0}{\log N}\longrightarrow0 .
\]
\end{proof}
\uses{Lemmas~\ref{lem:dickman}, \ref{lem:psi-upper}.}

\section{The upper bound: fibres of the rough part}\label{sec:upper}
%======================================================================

For $n\ge1$ let $R_B(n):=\prod_{p^v\|n,\ p>B}p^v$ be the $B$-rough part of $n$.

\begin{lemma}[Fibres]\label{lem:fibres}\status{PROVEN; essentially known, cf.~\cite{McN}}\lean{cofactor_le_B, roughPart_eq_of_edge, roughPart_eq_of_reachable, card_fibre_le, card_fibre_le_psi, fibre_one, card_smoothFibre, card_fibre_le_K}
Let $0\le K<N$ and $B=\lfloor N/(K+1)\rfloor$.
\begin{enumerate}[label=\textup{(\alph*)}]
\item If $\{d,m\}$ is an edge of $G_{N,K}$ with $d<m$, then $m/d\le B$, and hence $R_B(d)=R_B(m)$.
\item $R_B$ is constant on every connected component of $G_{N,K}$.
\item For $n\ge1$ let $V_n:=\{v\in V_{N,K}:\ R_B(v)=n\}$. Then $|V_n|\le\Psi(N/n,B)\le\Psi(N,B)$. Moreover $V_1=F_1$, $|F_1|=\Psi(N,B)-\Psi(K,B)$, and $|V_n|\le K$ for $n>1$.
\end{enumerate}
\end{lemma}
\begin{proof}
(a) Since $d\ge K+1$, we have $m/d\le N/(K+1)$. As $m/d$ is an integer, $m/d\le B$. So every prime factor of $m/d$ is at most $B$, and $v_p(m)=v_p(d)$ for every $p>B$.

(b) This follows from (a) by induction along paths.

(c) Each $v\in V_n$ factors uniquely as $v=n s$ with $s=v/n$ being $B$-smooth. Since $s\le N/n$, the map $v\mapsto s$ injects $V_n$ into $\{s\le N/n:\ \Pp(s)\le B\}$. Clearly $V_1=\{v\in(K,N]:\Pp(v)\le B\}=F_1$, and $|F_1|=\Psi(N,B)-\Psi(K,B)$. If $n>1$, then $n$ has a prime factor $>B$, so $n\ge B+1>N/(K+1)$. Hence $|V_n|\le N/n<K+1$.
\end{proof}
\uses{nothing.}

\begin{proposition}[Upper bound]\label{prop:upper}\status{PROVEN}\lean{stronglyConnected_subset_fibre, stronglyConnected_card_le_psi, stronglyConnected_card_le_max, PhiK_le_psi, PhiK_le_max}
For every orientation of $G_{N,K}$, every SCC is contained in a single fibre $V_n$. Hence
\[
\Phi_K\le\Psi(N,B)\qquad\text{and}\qquad\Phi_K\le\max(|F_1|,K).
\]
\end{proposition}
\begin{proof}
An SCC is connected in the underlying undirected graph, so it lies in a connected component of $G_{N,K}$, and hence in a fibre by Lemma~\ref{lem:fibres}(b). Now apply Lemma~\ref{lem:fibres}(c).
\end{proof}
\uses{Lemma~\ref{lem:fibres}.}

\begin{corollary}\label{cor:upper-profile}\status{PROVEN}\lean{cor_upper_profile, cor_upper_profile_eventually}
Let $\rho\in[0,1]$ be arbitrary. If $0\le K(N)<N$ for all large $N$ and $\log(K+1)/\log N\to\theta$, then $\limsup_N\E[\Phi_K]/N\le\dick(1/(1-\theta))$, with $\dick(\infty)=0$.
\end{corollary}
\begin{proof}
By Proposition~\ref{prop:upper}, $\E[\Phi_K]/N\le\Psi(N,N/(K+1))/N$, and the right side tends to $\dick(1/(1-\theta))$ by Lemma~\ref{lem:psi-limit}.
\end{proof}

\begin{remark}[Equivalent forms]\label{rem:F4}\status{PROVEN}\lean{remark_3_4, remark_3_4_holds}
Let $\rho\in[0,1]$ and suppose $K=o(N)$. The following are equivalent: (i) $\E[\Phi_K]=\Psi(N,B)+o(N)$; (ii) $\E\,\bigl||F_1|-\Phi_K\bigr|=o(N)$; (iii) for every $\varepsilon>0$, w.h.p.\ the largest SCC has at least $|F_1|-\varepsilon N$ vertices. Theorem~\ref{thm:main} establishes (i), and hence all three forms, for $\rho\in(0,1)$.

\emph{Proof.} Pointwise, $|F_1|\le\Psi(N,B)\le|F_1|+K$ by Lemma~\ref{lem:fibres}(c), and $\Phi_K\le\Psi(N,B)$ and $\Phi_K\le\max(|F_1|,K)$ by Proposition~\ref{prop:upper}. Hence $\bigl||F_1|-\Phi_K\bigr|\le(\Psi(N,B)-\Phi_K)+K$, and taking expectations gives (i)$\Rightarrow$(ii). Markov's inequality gives (ii)$\Rightarrow$(iii). Finally, $\Psi(N,B)-\Phi_K\le K+\varepsilon N+N\cdot\mathbf 1[\Phi_K<|F_1|-\varepsilon N]$, which gives (iii)$\Rightarrow$(i).\qed
\end{remark}

\section{The lower bound}\label{sec:lower}
%======================================================================

Throughout this section $N\ge2$, $0\le K<N$ and $\delta\in(0,\frac18]$ are fixed, and
\[
M:=N^{1-2\delta}\ \ (\text{a real number}),\qquad Y:=\frac{M}{2(K+1)},\qquad S_M:=\{x\in\Z\cap(K,M]:\ \Pp(x)\le Y\}.
\]
We write $G_{M,K}$ for the induced subgraph of $G_N$ on $\Z\cap(K,M]$. This is a subgraph of $G_{N,K}$, and it is not random.

\subsection{Robust edges}

\begin{lemma}[Robust edges]\label{lem:robust}\status{PROVEN; new}\lean{prob_not_robust, prob_not_robust_exp, robust_pair, robust_adj, robust_reachable, card_multiples_Ioc_ge}
Let $\cR$ be the event that for all integers $x,x'$ with $K<x<x'\le M$ and $x\mid x'$ there are vertices $y,y'\in V_{N,K}$, both multiples of $x'$, such that $x\to y\to x'$ and $x'\to y'\to x$ in $\cD_\rho(N,K)$. Then
\[
\Prob(\cR^c)\le M^2(1-q)^{N^{2\delta}-2}\le N^2\exp\bigl(-q(N^{2\delta}-2)\bigr).
\]
On $\cR$, the two endpoints of every edge of $G_{M,K}$ lie in the same SCC of $\cD_\rho(N,K)$. Consequently every connected component of $G_{M,K}$ lies in a single SCC. (The formal version allows any real $M\in(0,N]$, with $N/M$ in place of $N^{2\delta}$.)
\end{lemma}
\begin{proof}
Fix $x,x'$ as in the statement and let $\mathcal Y:=\{y\in\Z:\ x'<y\le N,\ x'\mid y\}$. Then
\[
|\mathcal Y|=\lfloor N/x'\rfloor-1\ge N/M-2=N^{2\delta}-2 .
\]
For $y\in\mathcal Y$ we have $K<x<x'<y\le N$ and $x\mid x'\mid y$. So $\{x,y\}$ and $\{x',y\}$ are edges of $G_{N,K}$, and their default orientations are $y\to x$ and $y\to x'$. Let $E_y$ be the event $\{x\to y\text{ and }y\to x'\}$, which says that $\{x,y\}$ is reversed and $\{x',y\}$ is not. Then $\Prob(E_y)=\rho(1-\rho)=q$. For distinct $y$ the pairs of edges involved are disjoint, so the events $E_y$, $y\in\mathcal Y$, are independent. Hence
\[
\Prob\Bigl(\bigcap_{y\in\mathcal Y}E_y^c\Bigr)=(1-q)^{|\mathcal Y|}\le(1-q)^{N^{2\delta}-2}.
\]
The same bound holds for the events $E'_y=\{x'\to y\text{ and }y\to x\}$, which say that $\{x',y\}$ is reversed and $\{x,y\}$ is not. The pairs $(x,x')$ are the edges of $G_{M,K}$, and $2|E(G_{M,K})|\le\lfloor M\rfloor^2$. Each pair has two directions, so the union bound gives $\Prob(\cR^c)\le M^2(1-q)^{N^{2\delta}-2}$. The second inequality uses $M\le N$ and $(1-q)^t\le e^{-qt}$ for $t=N^{2\delta}-2\ge0$. If $N^{2\delta}<2$, the right-hand side exceeds $1$ and there is nothing to prove.

On $\cR$, if $\{x,x'\}$ is an edge of $G_{M,K}$ with $x<x'$, then $K<x<x'\le M$ and $x\mid x'$. So directed paths $x\to y\to x'$ and $x'\to y'\to x$ exist, and $x,x'$ lie in the same SCC. Lying in the same SCC is an equivalence relation, so it propagates along paths of $G_{M,K}$.
\end{proof}
\uses{independence of distinct edges; union bound.}

\noindent\emph{Referee check.} (1) Independence is used only for the events $E_y$, $y\in\mathcal Y$, for a \emph{fixed} pair $(x,x')$; these involve pairwise disjoint sets of edges. Across different pairs no independence is used, only the union bound. (2) The bound $|\mathcal Y|\ge N^{2\delta}-2$ holds for every $x'\le M$, so the union bound is uniform. (3) The vertices $y$ may lie above $M$; this is harmless, since paths may use any vertex of $V_{N,K}$. (4) If $K=0$, the vertex $1$ is included and the argument is unchanged.

\subsection{Doubling}

\begin{lemma}[Doubling]\label{lem:doubling}\status{PROVEN; new in this form, elementary}\lean{doubling, doubling_connected, reachable_two_pow_mul}
Assume $M\ge4(K+1)$, so that $Y\ge2$. Then every $x\in S_M$ is joined in $G_{M,K}$ to the largest power of $2$ not exceeding $M$. In particular $S_M$ lies in a single connected component of $G_{M,K}$.
\end{lemma}
\begin{proof}
Let $2^b$ be the power of $2$ with $M/2<2^b\le M$. Then $2^b>M/2\ge 2(K+1)>K$ and $\Pp(2^b)=2\le Y$, so $2^b\in S_M$ (in particular $S_M\ne\emptyset$, which is not needed below). For $x\in S_M$ let $\Omega'(x)$ be the number of odd prime factors of $x$ counted with multiplicity. We show by induction on $\Omega'(x)$ that $x$ is joined to $2^b$ by a path in $G_{M,K}$ that stays inside $S_M$.

Let $a\ge0$ be maximal with $2^ax\le M$, and put $y:=2^ax$. Then $M/2<y\le M$. The vertices $x,2x,\dots,2^ax=y$ lie in $(K,M]$, each divides the next, and they all lie in $S_M$ (their odd part is that of $x$, and $2\le Y$). So they form a path in $G_{M,K}$. If $\Omega'(x)=0$, then $y$ is a power of $2$ in $(M/2,M]$, so $y=2^b$. Otherwise let $p$ be an odd prime dividing $x$, and put $x_1:=y/p$. Then $x_1\mid y$, $x_1<y\le M$, and
\[
x_1=\frac{y}{p}>\frac{M/2}{Y}=K+1 ,
\]
using $p\le\Pp(x)\le Y$. So $\{x_1,y\}$ is an edge of $G_{M,K}$. Moreover $\Pp(x_1)\le\max(\Pp(x),2)\le Y$, so $x_1\in S_M$, and $\Omega'(x_1)=\Omega'(x)-1$. By induction $x_1$ is joined to $2^b$, hence so is $x$.
\end{proof}
\uses{nothing.}

\begin{corollary}\label{cor:Sigma}\status{PROVEN}\lean{sigma_mutual, goodSet_mem_smoothFibre, sigma_subset_smoothFibre, sigmaSet_mutual, sigmaSet_closed, sigmaSet_subset}
Assume $M\ge4(K+1)$, and let $\Sigma$ be the set of vertices $v\in V_{N,K}$ that are mutually reachable in $\cD_\rho(N,K)$ with some element of $S_M$. On $\cR$, any two elements of $S_M$ lie in the same SCC. Hence, on $\cR$, $\Sigma$ is strongly connected and closed under mutual reachability, so it is the SCC containing $S_M$ (recall $S_M\ne\emptyset$). Since $M\le N$, for every outcome $\Sigma\subset F_1$.
\end{corollary}
\begin{proof}
The first claim follows from Lemmas~\ref{lem:robust} and~\ref{lem:doubling}. If $a,b\in\Sigma$ are mutually reachable with $d,d'\in S_M$, then $a\sim d\sim d'\sim b$ on $\cR$. Closure under mutual reachability is immediate from the definition. For $x\in S_M$ we have $\Pp(x)\le Y<N/(K+1)$, so $\Pp(x)\le B$ and $S_M\subset F_1=V_1$. If $v\in\Sigma$ is mutually reachable with $d\in S_M$, then $\{v,d\}$ is strongly connected, so it lies in a single fibre by Proposition~\ref{prop:upper}, which must be $V_1$. Hence $v\in F_1$.
\end{proof}

\subsection{Attachment}

For $m\in V_{N,K}$ with $m>M$ let $D(m):=\{d\in S_M:\ d\mid m\}$ and $s(m):=|D(m)|$. Let $A_m$ be the event that there are $d,d'\in D(m)$ with $m\to d$ and $d'\to m$.

\begin{lemma}[Attachment]\label{lem:attach}\status{PROVEN}\lean{prob_attachEv, prob_attachEv_ge, prob_all_bit, attach_mutual, prob_attach}
Assume $M\ge4(K+1)$ and let $m\in V_{N,K}$ with $m>M$. Then $\Prob(A_m)\ge1-\rho^{s(m)}-(1-\rho)^{s(m)}\ge1-2\lambda^{s(m)}$, with equality in the first step if $s(m)\ge1$. On $A_m\cap\cR$ we have $m\in\Sigma$.
\end{lemma}
\begin{proof}
Every $d\in D(m)$ satisfies $d\le M<m$, so the edges $\{d,m\}$, $d\in D(m)$, are $s(m)$ distinct edges of $G_{N,K}$ with independent orientations. The complement $A_m^c$ is contained in the union of the events ``all point $m\to d$'' and ``all point $d\to m$'', which are cylinders with probabilities $(1-\rho)^{s(m)}$ and $\rho^{s(m)}$; for $s(m)\ge1$ the union is disjoint and equals $A_m^c$. On $A_m\cap\cR$ we have $m\to d$, a directed path $d\rightsquigarrow d'$ in $\cD_\rho(N,K)$ (Corollary~\ref{cor:Sigma}, since $d,d'\in S_M$), and $d'\to m$. This is a closed walk through $m$ and $d$, so $m$ is mutually reachable with $d\in S_M$, that is, $m\in\Sigma$.
\end{proof}
\uses{Corollary~\ref{cor:Sigma}; independence of distinct edges.}

\noindent\emph{Referee check.} The events $A_m$ and $\cR$ are \emph{not} independent: the edges $\{d,m\}$ may be among those used in the definition of $\cR$. We never use independence between them. Below we only use $\Prob(A_m\cap\cR)\ge\Prob(A_m)-\Prob(\cR^c)$.

\subsection{The arithmetic core}

\begin{lemma}[Many divisors in the lower layer]\label{lem:core}\status{PROVEN; new}\lean{core_a, core_b, core_c, core_divisors, core_d, core_d'}
Let $\eta\in(0,1)$, let $s_0\ge1$ be an integer and $\delta\in(0,\eta/8]$, and put $z:=N^{\delta/s_0}$. Assume
\[
\textup{(H1)}\ K\le N^{1-\eta},\qquad \textup{(H2)}\ N^{\eta/4}\ge4,\qquad \textup{(H3)}\ z\ge2 .
\]
Let
\[
\cU:=\{m\in\Z:\ N^{1-\delta}<m\le N,\ \Pp(m)\le Y,\ \omega_z(m)\ge s_0\}.
\]
Then:
\begin{enumerate}[label=\textup{(\alph*)}]
\item $M\ge4(K+1)$;
\item $\cU\subset F_1$, and every $m\in\cU$ satisfies $m>M$;
\item $s(m)\ge s_0+1$ for every $m\in\cU$;
\item $\displaystyle |F_1\setminus\cU|\le N^{1-\delta}+N\Bigl(\frac{4\delta}{\eta}+\frac{2(1+C_1)}{\eta\log N}\Bigr)+\#\{m\le N:\ \omega_z(m)<s_0\}$.
\end{enumerate}
\end{lemma}
\begin{proof}
Note first that $K+1\le 2N^{1-\eta}$ by (H1), and that $\eta-2\delta\ge\frac34\eta$ and $\eta-6\delta\ge\frac14\eta$ because $\delta\le\eta/8$.

(a) $M/(K+1)\ge N^{1-2\delta}/(2N^{1-\eta})=\frac12N^{\eta-2\delta}\ge\frac12(N^{\eta/4})^3\ge32$, by (H2).

(b) Let $m\in\cU$. Then $m>N^{1-\delta}>N^{1-2\delta}=M$ and $m>N^{1-\delta}\ge N^{1-\eta}\ge K$, so $m\in V_{N,K}$. Also $\Pp(m)\le Y<N/(K+1)$, so $\Pp(m)\le B$ and $m\in F_1$.

(c) Fix $m\in\cU$ and put $E_0:=N^{2\delta}$ and $E_1:=N^{1-\delta}/(K+1)$.

\emph{Step 1: divisors in $[E_0,E_1]$ give elements of $D(m)$.} Let $e\mid m$ with $E_0\le e\le E_1$ and put $d:=m/e$. Then $d\le N/E_0=M$ and $d\ge m/E_1>N^{1-\delta}(K+1)/N^{1-\delta}=K+1$. Also $\Pp(d)\le\Pp(m)\le Y$. Hence $d\in S_M$ and $d\mid m$, so $d\in D(m)$. Distinct $e$ give distinct $d$.

\emph{Step 2: a reserve of small primes.} Since $\omega_z(m)\ge s_0$, we may choose distinct primes $p_1,\dots,p_{s_0}\le z$ dividing $m$. Put $r:=p_1\cdots p_{s_0}$, so $r\mid m$ and $r\le z^{s_0}=N^{\delta}$. Then $m':=m/r$ is an integer with $m'\ge m/N^\delta>N^{1-2\delta}$.

\emph{Step 3: a base divisor $e_0\mid m'$ with $E_0\le e_0\le E_1/z$.} First note that
\[
\frac{E_1}{z}=\frac{N^{1-\delta-\delta/s_0}}{K+1}\ge\frac{N^{1-2\delta}}{K+1}=2Y .
\]
\emph{Case (i): $m'$ has a prime factor $q\ge E_0$.} Then $E_0\le q\le\Pp(m)\le Y\le E_1/z$, and we take $e_0:=q$.

\emph{Case (ii): every prime factor of $m'$ is $<E_0$.} Write $m'=q_1q_2\cdots q_t$ with primes $q_i$ (with multiplicity, in any order), and let $\pi_i:=q_1\cdots q_i$. Then $\pi_0=1<E_0\le N^{1-2\delta}<m'=\pi_t$, where $E_0\le N^{1-2\delta}$ because $\delta\le\frac14$. Let $j$ be the least index with $\pi_j\ge E_0$. Then $j\ge1$ and $\pi_j=\pi_{j-1}q_j<E_0^2=N^{4\delta}$. Moreover
\[
\frac{E_1}{z}\ge\frac{N^{1-2\delta}}{K+1}\ge\tfrac12N^{\eta-2\delta}=N^{4\delta}\cdot\tfrac12N^{\eta-6\delta}\ge N^{4\delta}\cdot\tfrac12N^{\eta/4}\ge N^{4\delta},
\]
by (H2). So $e_0:=\pi_j$ satisfies $E_0\le e_0\le E_1/z$.

\emph{Step 4.} Since $e_0\mid m'$, we have $e_0r\mid m$, so $e_0p_i\mid m$ for each $i$. The $s_0+1$ numbers $e_0,e_0p_1,\dots,e_0p_{s_0}$ are distinct divisors of $m$, and they all lie in $[E_0,E_1]$ because $e_0p_i\le e_0z\le E_1$. By Step~1, $s(m)\ge s_0+1$.

(d) If $m\in F_1\setminus\cU$, then $m\le N^{1-\delta}$, or $Y<\Pp(m)\le B$, or $\omega_z(m)<s_0$. The first set has at most $N^{1-\delta}$ elements. The second is empty if $Y\ge B$. Otherwise its size is at most $\sum_{Y<p\le B}\lfloor N/p\rfloor\le N\sum_{Y<p\le B}\frac1p$. We have $Y\ge2$ by (a). Moreover
\[
\log B-\log Y\le\log\frac{N}{K+1}-\log\frac{N^{1-2\delta}}{2(K+1)}=2\delta\log N+\log2,
\]
and
\[
\log Y\ge\log\frac{N^{1-2\delta}}{4N^{1-\eta}}=(\eta-2\delta)\log N-\log4\ge\tfrac34\eta\log N-\log 4\ge\tfrac12\eta\log N,
\]
by (H2). By Lemma~\ref{lem:mertens}(b) with $C_1=18$ and $\log(1+t)\le t$,
\[
\sum_{Y<p\le B}\frac1p\le\frac{\log B-\log Y}{\log Y}+\frac{C_1}{\log Y}\le\frac{2\delta\log N+1+C_1}{\frac12\eta\log N}=\frac{4\delta}{\eta}+\frac{2(1+C_1)}{\eta\log N}.\qedhere
\]
\end{proof}
\uses{Lemma~\ref{lem:mertens}(b).}

\noindent\emph{Referee check.} (1) No ``almost all'' statement is used inside the proof. Every $m\in\cU$ has $s(m)\ge s_0+1$ deterministically, and the exceptional set is counted explicitly in (d). (2) The divisor $e_0$ is found by a deterministic case analysis. Case (i) covers a prime factor of any size in $[E_0,\Pp(m)]$, including sizes close to $Y$; this is where the reserve primes must be tiny ($\le z$), because $e_0$ may be as large as $Y$. (3) The bounds are uniform in $K\le N^{1-\eta}$; $K$ enters only through (H1).

\subsection{The lower bound}

\begin{proposition}[Lower bound]\label{prop:lower}\status{PROVEN}\lean{lower_bound, lower_bound_err', fixedN_sandwich', eq_4_2_core, eq_4_2}
Assume the hypotheses of Lemma~\ref{lem:core}, and put
\[
\mathrm{Err}:=N^{1-\delta}+N\Bigl(\frac{4\delta}\eta+\frac{2(1+C_1)}{\eta\log N}\Bigr)+\#\{m\le N:\omega_z(m)<s_0\}+2\lambda^{s_0+1}N .
\]
Then
\begin{align}
\E[\Phi_K]&\ge\Psi(N,B)-K-\mathrm{Err}-N\,\Prob(\cR^c),\label{eq:lower1}\\
\E\bigl[|F_1\setminus\Sigma|\,\mathbf 1_{\cR}\bigr]&\le\mathrm{Err}.\label{eq:lower2}
\end{align}
Here $C_1=18$, $\Sigma$ is as in Corollary~\ref{cor:Sigma}, and $\Prob(\cR^c)\le N^2\exp(-q(N^{2\delta}-2))$ by Lemma~\ref{lem:robust}.
\end{proposition}
\begin{proof}
By Lemma~\ref{lem:core}(a), Corollary~\ref{cor:Sigma} and Lemma~\ref{lem:attach} apply. Let $X:=\mathbf 1_{\cR}\cdot\#\{m\in\cU:\ A_m\text{ occurs}\}$. On $\cR$, every $m\in\cU$ for which $A_m$ occurs lies in $\Sigma$ (Lemmas~\ref{lem:core}(b) and~\ref{lem:attach}), so $X\le|\Sigma|\le\Phi_K$. On $\cR^c$, $X=0\le\Phi_K$. By Lemmas~\ref{lem:core}(c) and~\ref{lem:attach},
\[
\E[\Phi_K]\ge\E[X]=\sum_{m\in\cU}\Prob(A_m\cap\cR)\ge\sum_{m\in\cU}\bigl(\Prob(A_m)-\Prob(\cR^c)\bigr)\ge|\cU|\bigl(1-2\lambda^{s_0+1}\bigr)-N\Prob(\cR^c).
\]
Moreover $|\cU|\ge|F_1|-|F_1\setminus\cU|$ and $|F_1|=\Psi(N,B)-\Psi(K,B)\ge\Psi(N,B)-K$. Also $2\lambda^{s_0+1}|\cU|\le2\lambda^{s_0+1}N$. Combining these with Lemma~\ref{lem:core}(d) proves~\eqref{eq:lower1}. For~\eqref{eq:lower2}: on $\cR$, every $m\in\cU$ for which $A_m$ occurs lies in $\Sigma$ (Lemma~\ref{lem:attach}), so $F_1\setminus\Sigma\subset(F_1\setminus\cU)\cup\{m\in\cU:\ A_m^c\}$. Taking cardinalities pointwise, $|F_1\setminus\Sigma|\,\mathbf 1_\cR\le|F_1\setminus\cU|+\sum_{m\in\cU}\mathbf 1_{A_m^c}$ for every outcome, and $\Prob(A_m^c)\le2\lambda^{s_0+1}$ by Lemmas~\ref{lem:core}(c) and~\ref{lem:attach}. So
\[
\E\bigl[|F_1\setminus\Sigma|\mathbf 1_\cR\bigr]\le|F_1\setminus\cU|+\sum_{m\in\cU}\Prob(A_m^c)\le|F_1\setminus\cU|+2\lambda^{s_0+1}N\le \mathrm{Err}.\qedhere
\]
\end{proof}
\uses{Lemmas~\ref{lem:robust}, \ref{lem:doubling}, \ref{lem:attach}, \ref{lem:core}; Corollary~\ref{cor:Sigma}.}

\noindent\emph{Referee check.} (1) The only probabilistic inputs are exact computations for independent edges (Lemmas~\ref{lem:robust} and~\ref{lem:attach}), the union bound, and $\Prob(A\cap\cR)\ge\Prob(A)-\Prob(\cR^c)$. (2) The inequality $\Phi_K\ge X$ holds pointwise on the whole probability space. (3) Proposition~\ref{prop:lower} is a statement for each fixed $N$, with no limits, so there is no interchange of limits. Uniformity in $K\le N^{1-\eta}$ is explicit. Note that the term $\#\{m\le N:\omega_z(m)<s_0\}$ does not depend on $K$.

\begin{proof}[Proof of Theorem~\ref{thm:rate}]
The upper bound $\Phi_K\le\Psi(N,B)$ is Proposition~\ref{prop:upper}. For the lower bound, let $N$ be large and put
\[
\ell:=\log\log N,\qquad\eta:=\frac1\ell,\qquad\delta:=\frac1{8\ell^2},\qquad s_0:=\max\Bigl(1,\Bigl\lceil\frac{\log\ell}{\log(1/\lambda)}\Bigr\rceil\Bigr),\qquad z:=N^{\delta/s_0}.
\]
Then $\delta\le\eta/8$ and $\lambda^{s_0}\le 1/\ell$. Write $t:=\log N$, so $\ell=\log t$. Since $s_0=O_\rho(\log\ell)$, and $A(\log t)^k\le t$ for large $t$ for every fixed $A$ and $k$, there is $N_0(\rho)$ such that for $N\ge N_0(\rho)$ all of the following hold:
\begin{enumerate}[label=(R\arabic*)]
\item $\ell>1$, $N^{\eta/4}=\exp\bigl(\frac{t}{4\ell}\bigr)\ge4$ and $z=\exp\bigl(\frac{t}{8\ell^2s_0}\bigr)\ge2$;
\item $L:=\sum_{p\le z}\frac1p\ \ge\ \log\log z-13=\ell-\log(8\ell^2s_0)-13\ \ge\ \ell/2+2s_0$ (Lemma~\ref{lem:mertens}(c)), so $L\ge\max(2s_0,\ell/2)$;
\item $N^{1-\eta}\le N/\ell$ and $N^{1-\delta}\le N/\ell$, that is, $\ell\log\ell\le t$ and $8\ell^2\log\ell\le t$;
\item $2(1+C_1)/(\eta\log N)=38\ell/t\le1/(2\ell)$, that is, $76\ell^2\le t$;
\item $N\cdot N^2\exp(-q(N^{2\delta}-2))\le1$, that is, $3t\le q\bigl(\exp(t/(4\ell^2))-2\bigr)$, which holds because $N^{2\delta}=\exp(t/(4\ell^2))$ grows faster than any power of $t$;
\item $\sqrt N\le N/(2\ell)$ and $2c_0N/\log N\le N/(2\ell)$.
\end{enumerate}
These are the conditions \texttt{GoodT} of the formalisation (\texttt{eventually\_goodN}). Note also $N/\ell\ge1$.
\emph{Case $K\le N^{1-\eta}$.} Hypotheses (H1)--(H3) of Lemma~\ref{lem:core} hold by (R1). By Lemma~\ref{lem:turan} and (R2), $\#\{m\le N:\omega_z(m)<s_0\}\le12N/L\le24N/\ell$. Also $4\delta/\eta=1/(2\ell)$ and $2\lambda^{s_0+1}\le2/\ell$. Proposition~\ref{prop:lower} together with (R3)--(R5) gives
\[
\Psi(N,B)-\E[\Phi_K]\le\frac N\ell+\frac N\ell+N\Bigl(\frac1{2\ell}+\frac1{2\ell}\Bigr)+\frac{24N}\ell+\frac{2N}\ell+1\le\frac{29N}{\ell}+1\le\frac{30N}{\ell}.
\]
In this case the terms of $\mathrm{Err}$ satisfy $\mathrm{Err}\le 28N/\ell$, and $N\Prob(\cR^c)\le1$; we use this again in the proof of Corollary~\ref{cor:structure}.

\emph{Case $K>N^{1-\eta}$.} Then $B\le N/(K+1)<N^{\eta}$, so $\log B<\eta\log N=t/\ell$. Since $K<N$, $B\ge1$, and Lemma~\ref{lem:psi-upper} and (R6) give
\[
\Psi(N,B)\le\sqrt N+\frac{2N\log B}{\log N}+\frac{2c_0N}{\log N}\le\frac N{2\ell}+\frac{2N}\ell+\frac N{2\ell}=\frac{3N}\ell .
\]
Hence $0\le\Psi(N,B)-\E[\Phi_K]\le\Psi(N,B)\le 30N/\ell$.
\end{proof}
\uses{Propositions~\ref{prop:upper}, \ref{prop:lower}; Lemmas~\ref{lem:mertens}, \ref{lem:psi-upper}, \ref{lem:turan}, \ref{lem:core}.}

\noindent\emph{Referee check.} (1) The parameters $\eta,\delta,s_0,z$ depend on $N$ only, not on $K$. So the two cases cover all $K$ with one common bound. (2) The threshold $N_0(\rho)$ depends on $\rho$ through $s_0$ (via $\lambda$) and through $q$ in (R5). (3) Proposition~\ref{prop:lower} is applied at each fixed $N$, so no limit is interchanged.

\begin{proof}[Proof of Theorem~\ref{thm:main}]
The deterministic bound is Proposition~\ref{prop:upper}. The limit follows from Theorem~\ref{thm:rate}, since $30/\log\log N\to0$.
\end{proof}

\begin{remark}[Explicit form for $K\le\sqrt N-1$]\label{rem:smallK}\status{PROVEN}\lean{remark_5_1, smallK_psi, N_sub_psi_eq}
For $N\ge N_0'(\rho)$ and $0\le K\le\sqrt N-1$,
\[
\Bigl|1-\frac{\E[\Phi_K]}N-\log\frac{\log N}{\log B}\Bigr|\le\frac{31}{\log\log N}.
\]
\emph{Proof.} Since $N/(K+1)\ge\sqrt N$, we have $B\ge\lfloor\sqrt N\rfloor$, and every prime $p>B$ satisfies $p>\sqrt N$. So each $n\le N$ has at most one prime factor $>B$, and that factor divides $n$ exactly once. Hence $N-\Psi(N,B)=\sum_{B<p\le N}\lfloor N/p\rfloor$, which differs from $N\sum_{B<p\le N}\frac1p$ by at most $\pi(N)$. By Lemma~\ref{lem:mertens}(b),(d) and $\log B\ge\frac14\log N$ (for $N\ge16$),
\[
\Bigl|1-\frac{\Psi(N,B)}N-\log\frac{\log N}{\log B}\Bigr|\le\frac{C_1}{\log B}+\frac{\pi(N)}N\le\frac{4C_1+5}{\log N}=\frac{77}{\log N}\le\frac1{\log\log N}
\]
for large $N$. Combine this with $0\le(\Psi(N,B)-\E[\Phi_K])/N\le30/\log\log N$ (Theorem~\ref{thm:rate}).\qed
\end{remark}

\begin{proof}[Proof of Corollary~\ref{cor:profile}]
Write $\E[\Phi_K]/N=\Psi(N,N/(K+1))/N-\mathrm{gap}/N$ with $\mathrm{gap}:=\Psi(N,N/(K+1))-\E[\Phi_K]$. By Theorem~\ref{thm:rate}, $0\le\mathrm{gap}/N\le30/\log\log N\to0$, and $\Psi(N,N/(K+1))/N\to\dick(1/(1-\theta))$ by Lemma~\ref{lem:psi-limit}.

Write $P(\theta)$ for the profile. On $(-\infty,1)$, $P$ is continuous because $\dick$ and $\theta\mapsto1/(1-\theta)$ are; on $(1,\infty)$ it is $0$. At $\theta=1$ it is continuous because $0\le P(\theta)\le2(1-\theta)$ for $\theta\in[0,1)$, by $\dick(u)\le2/u$ (Lemma~\ref{lem:psi-upper}). On $[0,1)$, $P$ is strictly decreasing because $\theta\mapsto1/(1-\theta)$ is strictly increasing with values in $[1,\infty)$, where $\dick$ is strictly decreasing (Lemma~\ref{lem:dickman}); and $P(1)=0<P(\theta)$ for $\theta<1$ since $\dick>0$. For $0\le\theta\le\frac12$ we have $u=1/(1-\theta)\in[1,2]$, where $\dick(u)=1-\log u$.
\end{proof}
\uses{Theorem~\ref{thm:main}; Lemmas~\ref{lem:dickman}, \ref{lem:psi-upper}, \ref{lem:psi-limit}.}

\begin{proof}[Proof of Corollary~\ref{cor:structure}]
Use the parameters and $N_0(\rho)$ from the proof of Theorem~\ref{thm:rate}. The SCCs are the classes of mutual reachability; for $K<N$ a largest one exists, so rules $\Phi^*_K$ exist. Write $S:=\Phi^*_K$, so $|S|=\Phi_K$.

(a) By Proposition~\ref{prop:upper}, $|\Phi_K-\Psi(N,B)|=\Psi(N,B)-\Phi_K$ pointwise, so $\E|\Phi_K-\Psi(N,B)|=\Psi(N,B)-\E[\Phi_K]\le30N/\ell$ by Theorem~\ref{thm:rate}. For $\varepsilon>0$, Markov's inequality gives $\Prob\bigl(|\Phi_K/N-\Psi(N,B)/N|\ge\varepsilon\bigr)\le30/(\varepsilon\ell)$ for every $K<N$, and this tends to $0$.

(b), (c): \emph{Case $K\le N^{1-\eta}$.} Then the hypotheses of Lemma~\ref{lem:core} hold, so $M\ge4(K+1)$ and $M\le N$. Work on $\cR$. By Corollary~\ref{cor:Sigma}, $\Sigma\subset F_1$ is strongly connected and closed under mutual reachability. Hence every SCC $C$ either equals $\Sigma$ or is disjoint from it: if $x\in C\cap\Sigma$, then $C=\Sigma$. Every SCC lies in a fibre $V_n$ (Proposition~\ref{prop:upper}); if $n=1$ then $C\subset F_1$, and if $n>1$ then $|C|\le K$ (Lemma~\ref{lem:fibres}(c)). So every SCC $C\ne\Sigma$ has $|C|\le W+K$, where $W:=|F_1\setminus\Sigma|$. Also $|\Sigma|\le\Phi_K=|S|$, and $|F_1|=W+|\Sigma|$.
\begin{itemize}
\item If $S=\Sigma$, then $|F_1\triangle S|=|F_1\setminus\Sigma|=W$.
\item If $S\ne\Sigma$ and $S\subset F_1$, then $|F_1\triangle S|=|F_1|-|S|\le|F_1|-|\Sigma|=W$.
\item If $S\ne\Sigma$ and $|S|\le K$, then $|F_1\triangle S|\le|F_1|+|S|=W+|\Sigma|+|S|\le W+2K$.
\end{itemize}
For $\Phi^{(2)}_K$, let $C\ne S$ be an SCC. If $C\ne\Sigma$, then $|C|\le W+K$. If $C=\Sigma$, then $S\ne\Sigma$, so $|C|=|\Sigma|\le|S|\le W+K$.

Thus, on $\cR$, $|F_1\triangle S|\le W+2K$ and $\Phi^{(2)}_K\le W+K$. For every outcome both quantities are at most $N$, since all sets involved lie in $V_{N,K}$. Hence
\[
\E|F_1\triangle S|\le2K+\E[W\mathbf 1_\cR]+N\Prob(\cR^c),\qquad\E[\Phi^{(2)}_K]\le K+\E[W\mathbf 1_\cR]+N\Prob(\cR^c).
\]
By~\eqref{eq:lower2} and the estimates in the proof of Theorem~\ref{thm:rate}, $\E[W\mathbf 1_\cR]\le\mathrm{Err}\le 28N/\ell$ and $N\Prob(\cR^c)\le1\le N/\ell$. Hence $\E|F_1\triangle S|\le 2K+29N/\ell$ and $\E[\Phi^{(2)}_K]\le K+29N/\ell$ in this case.

\emph{Case $K>N^{1-\eta}$.} Here $|F_1\triangle S|\le|F_1|+|S|\le2\Psi(N,B)\le6N/\ell$, using $|F_1|\le\Psi(N,B)$ (Lemma~\ref{lem:fibres}(c)) and $|S|=\Phi_K\le\Psi(N,B)$; and $\Phi^{(2)}_K\le\Phi_K\le\Psi(N,B)\le3N/\ell$.
\end{proof}
\uses{Propositions~\ref{prop:upper}, \ref{prop:lower}; Corollary~\ref{cor:Sigma}; Lemma~\ref{lem:fibres}; Theorem~\ref{thm:rate}.}

\begin{remark}[The term $K$ in Corollary~\ref{cor:structure}(c) is needed]\label{rem:second}\status{PROVEN}\lean{remark_5_2, secondSCC_ge, expect_secondSCC_ge, prob_PhiK_le_K}
Fix $\rho\in(0,1)$ and let $K=K(N)$ satisfy $K\to\infty$ and $K\le\sqrt N/2$. Then, for any rule choosing a largest SCC, $\liminf_{N\to\infty}\E[\Phi^{(2)}_K]/K\ge\frac12$. (All $o(1)$ terms below are as $N\to\infty$ along this sequence and depend on $\rho$ and on the sequence.)

\emph{Proof.} Let $N\ge16$. Then $(K+1)^2\le N$, and $B\ge\sqrt N$ because $N<(K+1)(B+1)$ and $K+1\le\sqrt N/2+1$. Let $p$ be a prime in $(B,2B]$, which exists by Bertrand's postulate. For $1\le s\le N/p$ we have $s<N/B\le\sqrt N\le B<p$. So $s$ is $B$-smooth and $p\nmid s$, which gives $R_B(ps)=p$. Also $ps\ge p>K$. Together with Lemma~\ref{lem:fibres}(c) this shows $V_p=\{ps:\ 1\le s\le N'\}$ with $N':=\lfloor N/p\rfloor$. Since $ps\mid ps'$ if and only if $s\mid s'$, the map $s\mapsto ps$ is an isomorphism from $G_{N'}$ onto $G_N[V_p]$ that preserves the default orientation. Hence the restriction of $\cD_\rho(N,K)$ to $V_p$ has the law of $\cD_\rho(N')$. As $V_p$ is a union of connected components of $G_{N,K}$, its SCCs are SCCs of $\cD_\rho(N,K)$. We have $K/2\le N'\le K$, because $N<p(N'+1)\le2B(N'+1)$, $B(K+1)\le N$ and $pN'\le N<(K+1)(B+1)\le(K+1)p$. So $N'\to\infty$. So by~\cite[Corollary~2]{KP} (or by Theorem~\ref{thm:main} with $K=0$, since $\Psi(N',N')=N'$) the largest SCC inside $V_p$ has expected size $(1-o(1))N'$.

Next, $\log B\ge\frac12\log N$, so the estimate in the proof of Remark~\ref{rem:smallK} gives $\Psi(N,B)/N\ge1-\log2-77/\log N\ge\frac14$ for large $N$. Also $K\le\sqrt N/2\le N/8$. By Theorem~\ref{thm:rate} and Markov's inequality,
\[
\Prob(\Phi_K\le K)\le\frac{\E[\Psi(N,B)-\Phi_K]}{\Psi(N,B)-K}\le\frac{30N/\log\log N}{N/8}=\frac{240}{\log\log N}\to0 .
\]
On the event $\{\Phi_K>K\}$ the largest SCC lies in $F_1=V_1$ (Lemma~\ref{lem:fibres}(c)), which is disjoint from $V_p$, so $\Phi^{(2)}_K\ge\#\Phi(\cD_\rho(N,K)[V_p])$. Therefore
\[
\E[\Phi^{(2)}_K]\ge\E\bigl[\#\Phi(\cD_\rho(N,K)[V_p])\bigr]-N'\,\Prob(\Phi_K\le K)=(1-o(1))N'\ge(\tfrac12-o(1))K.\qquad\qed
\]
\end{remark}

\begin{remark}[Uniformity in $\rho$; slowly vanishing $\rho$]\label{rem:rho}\status{PROVEN}\lean{remark_5_3_uniform, remark_5_3_uniform_main, GoodT.mono, remark_5_3_slow, slow_rho_bound}
The proof depends on $\rho$ only through $q=\rho(1-\rho)$ and $\lambda=\max(\rho,1-\rho)$. Let $0<\rho_0\le\frac12$. For $\rho\in[\rho_0,1-\rho_0]$ we have $\lambda\le\lambda_0:=1-\rho_0$ and $q\ge q_0:=\rho_0(1-\rho_0)$. Then $s_0(\lambda)\le s_0(\lambda_0)$. Conditions (R1)--(R4) and (R6) involve only $N$ and $s_0$, and they become weaker as $s_0$ decreases. Condition (R5) becomes weaker as $q$ increases, because its left side is decreasing in $q$ when $N^{2\delta}\ge2$. So (R1)--(R6) at $\rho_0$ imply (R1)--(R6) at every $\rho\in[\rho_0,1-\rho_0]$. Hence Theorems~\ref{thm:main} and~\ref{thm:rate} hold uniformly for $\rho\in[\rho_0,1-\rho_0]$, with $N_0=N_0(\rho_0)$. More generally, Theorem~\ref{thm:main} remains true for $\rho=\rho_N$ provided $\min(\rho_N,1-\rho_N)\log\log N\to\infty$. This follows from a quantitative form: for every $\varepsilon\in(0,1]$ there is $N_0(\varepsilon)$ such that $\Psi(N,N/(K+1))-\E[\Phi_K]\le\varepsilon N$ for all $N\ge N_0(\varepsilon)$, all $0\le K<N$ and all $\rho\in(0,1)$ with $\min(\rho,1-\rho)\log\log N\ge8\log(16/\varepsilon)+8$. To prove it, write $\ell=\log\log N$ and apply Proposition~\ref{prop:lower} with $\eta=\varepsilon/4$, $\delta=\varepsilon\eta/32\le\eta/8$ and $s_0:=\lfloor\ell/4\rfloor$, so that $\ell/8\le s_0\le\ell/4$. Each of the seven error terms is at most $\varepsilon N/8$ for large $N$. In particular, Lemma~\ref{lem:mertens}(c) gives $L\ge\log\delta+\ell-\log s_0-13\ge\ell/2$, so the $\omega_z$ term is at most $12N/L\le24N/\ell$. Next, $2\lambda^{s_0+1}\le2\exp(-\min(\rho,1-\rho)(s_0+1))\le\varepsilon/8$. Finally $N\Prob(\cR^c)\le N^3\exp(-q(N^{2\delta}-2))\le\varepsilon N/8$, since $q\ge\frac12\min(\rho,1-\rho)\ge1/(2\ell)$. The case $K>N^{1-\eta}$ is handled by Lemma~\ref{lem:psi-upper} as before.
\end{remark}

\section{Removing the vertices of highest degree}\label{sec:degree}
%======================================================================

For $T\subset[N]$ write $\deg_T(v)$ for the degree of $v$ in $G_N-T$. In $G_N$,
\[
\deg(m)=\bigl(\tau(m)-1\bigr)+\bigl(\lfloor N/m\rfloor-1\bigr).
\]
Put $\tau_N:=\max_{n\le N}\tau(n)$. Then $\deg_T(m)\le\deg(m)\le\tau_N+N/m$ for every $T$ and $m\notin T$. The proofs below use only this upper bound and the following lower bound: if every element of $T$ is at most $A$ and $1\le v\le A\le N$, then the multiples of $v$ in $(A,N]$ are neighbours of $v$ in $G_N-T$, so $\deg_T(v)\ge N/v-A/v-1$. \lean{deg_le, degT_le_deg, degT_ge}

\begin{lemma}[Monotonicity]\label{lem:mono}\status{PROVEN}\lean{maxSCC_mono, maxSCC_Ioc_antitone}
If $D$ is a directed graph and $U\subset V(D)$, then $\#\Phi(D[U])\le\#\Phi(D)$.
\end{lemma}
\begin{proof}
An SCC of $D[U]$ is strongly connected in $D$, so it is contained in an SCC of $D$.
\end{proof}

\begin{lemma}[Sandwich]\label{lem:sandwich}\status{PROVEN}\lean{sandwich_static, sandwich_adaptive}
Let $0<\varepsilon\le\frac12$ and $1\le K<N$ with $\frac{N}{K}\cdot\frac\varepsilon2\ge\hat\tau+5$, where $\hat\tau\ge\tau(n)$ for all $n\le N$ (for instance $\hat\tau=\tau_N$). Let $T$ be a static or an adaptive top-$K$ set as in Corollary~\ref{cor:degree}. Then
\[
\{1,\dots,\lfloor(1-\varepsilon)K\rfloor\}\subset T\subset\{1,\dots,\lfloor(1+\varepsilon)K\rfloor\}.
\]
\end{lemma}
\begin{proof}
\emph{Static, upper inclusion.} Suppose $w\in T$ with $w>(1+\varepsilon)K$. Then $|T\cap[1,K]|\le K-1$, so some $v\le K$ has $v\notin T$. Now
\[
\deg(v)\ge\lfloor N/v\rfloor-1\ge N/K-2\qquad\text{and}\qquad\deg(w)<\tau_N+N/((1+\varepsilon)K).
\]
The static condition gives $\deg(w)\ge\deg(v)$, hence $\frac NK\cdot\frac{\varepsilon}{1+\varepsilon}<\tau_N+2$. This contradicts the hypothesis, since $\frac\varepsilon{1+\varepsilon}\ge\frac\varepsilon2$.

\emph{Static, lower inclusion.} Suppose $v\le(1-\varepsilon)K$ and $v\notin T$. Every $t\in T$ satisfies $\tau_N+N/t\ge\deg(t)\ge\deg(v)\ge N/v-2\ge\frac{N}{(1-\varepsilon)K}-2$. If $t>(1-\frac\varepsilon2)K$, this gives $\frac NK\bigl(\frac1{1-\varepsilon}-\frac1{1-\varepsilon/2}\bigr)<\tau_N+2$. The left side is at least $\frac NK\cdot\frac\varepsilon2$, a contradiction. So $T\subset[1,(1-\frac\varepsilon2)K]$. But this set has fewer than $K$ elements, which contradicts $|T|=K$.

\emph{Adaptive, upper inclusion.} We show by induction on $j<K$ that $T_j:=\{u_1,\dots,u_j\}\subset[1,(1+\varepsilon)K]$ implies $u_{j+1}\le(1+\varepsilon)K$. Since $|T_j|<K$, some $v\le K$ has $v\notin T_j$. No multiple of $v$ in $((1+\varepsilon)K,N]$ has been removed, so
\[
\deg_{T_j}(v)\ge\frac{N-(1+\varepsilon)K}{v}-1\ge\frac NK-3 .
\]
Every $w>(1+\varepsilon)K$ has $\deg_{T_j}(w)<\tau_N+N/((1+\varepsilon)K)$, and this is at most $N/K-3$ by the hypothesis. Hence the maximiser $u_{j+1}$ satisfies $u_{j+1}\le(1+\varepsilon)K$.

\emph{Adaptive, lower inclusion.} Suppose $v\le(1-\varepsilon)K$ and $v\notin T_K$. For every $j<K$, using $T_j\subset[1,(1+\varepsilon)K]$,
\[
\deg_{T_j}(v)\ge\frac{N-(1+\varepsilon)K}{v}-1\ge\frac{N-2K}{(1-\varepsilon)K}-1 .
\]
Also $\tau_N+N/u_{j+1}\ge\deg(u_{j+1})\ge\deg_{T_j}(u_{j+1})\ge\deg_{T_j}(v)$. If $u_{j+1}>(1-\frac\varepsilon2)K$, then
\[
\frac NK\Bigl(\frac1{1-\varepsilon}-\frac1{1-\varepsilon/2}\Bigr)<\tau_N+1+\frac{2}{1-\varepsilon}\le\tau_N+5 .
\]
The left side is at least $\frac NK\cdot\frac\varepsilon2$, a contradiction. So all $K$ vertices $u_1,\dots,u_K$ lie in $[1,(1-\frac\varepsilon2)K]$, which has fewer than $K$ elements. This is impossible.
\end{proof}
\uses{the two degree bounds above, with $\hat\tau$ in place of $\tau_N$.}

\begin{proof}[Proof of Corollary~\ref{cor:degree}]
Let $\varepsilon=\frac14$ and $K_\pm:=\lfloor(1\pm\varepsilon)K\rfloor$.

\emph{Step 1: the coupling.} $\cD_\rho(N)-T$ is the subgraph of $\cD_\rho(N)$ induced on $[N]\setminus T$. For $K'<N$, the subgraph induced on $V_{N,K'}$ uses only the edges of $G_{N,K'}$, and their orientations are independent with the same law as in $\cD_\rho(N,K')$: the product measure on the edges of $G_N$ pushes forward to the product measure on the edges of $G_{N,K'}$. Hence $\E\bigl[\#\Phi(\cD_\rho(N)[V_{N,K'}])\bigr]=\E[\Phi_{K'}]$. \lean{expect_maxSCC_Ioc, expectP_comp_injective}

\emph{Step 2: the sandwich.} Put $\alpha:=(1-\theta)/2$ and $\beta:=(1+3\theta)/4$, so $\theta<\beta<1$ and $1-\beta=\alpha+\frac{1-\theta}4$. By Lemma~\ref{lem:divisor} with exponent $\alpha$, $\tau(n)\le C_\alpha N^\alpha=:\hat\tau$ for all $n\le N$. For large $N$ we have $\log(K+1)/\log N<\beta$, that is, $K+1<N^\beta$, so $N/K\ge N^{1-\beta}=N^\alpha N^{(1-\theta)/4}$. For large $N$ also $N^{(1-\theta)/4}\ge16C_\alpha$ and $N^{1-\beta}\ge80$. Then $\hat\tau+5\le N^{1-\beta}/16+N^{1-\beta}/16\le\frac NK\cdot\frac\varepsilon2$, which is the hypothesis of Lemma~\ref{lem:sandwich} when $K\ge1$. It gives $\{1,\dots,K_-\}\subset T\subset\{1,\dots,K_+\}$, that is, $V_{N,K_+}\subset[N]\setminus T\subset V_{N,K_-}$. If $K=0$, then $T=\emptyset$ and this is trivial.

\emph{Step 3.} Since $T$ does not depend on the orientation, Lemma~\ref{lem:mono} applied to the single random graph $\cD_\rho(N)$ gives $\#\Phi(\cD_\rho(N)[V_{N,K_+}])\le\#\Phi(\cD_\rho(N)-T)\le\#\Phi(\cD_\rho(N)[V_{N,K_-}])$ pointwise. By Step~1, $\E[\Phi_{K_+}]\le\E[\#\Phi(\cD_\rho(N)-T)]\le\E[\Phi_{K_-}]$.

\emph{Step 4.} We have $(K+1)/4\le K_-+1\le K+1\le K_++1\le2(K+1)$, so $|\log(K_\pm+1)-\log(K+1)|\le\log4$ and $\log(K_\pm+1)/\log N\to\theta$. Since $\theta<1$, $K_-\le K_+<N$ for large $N$. Corollary~\ref{cor:profile} applies to $K_\pm$, and both bounds divided by $N$ tend to $\dick(1/(1-\theta))$.

\emph{The last claim.} Let $C:=C_\kappa$ be the constant of Lemma~\ref{lem:divisor} with exponent $\kappa$, and let $N$ be so large that $x:=N^\kappa\ge2C+4$. Let $1\le K\le N^{1/2-\kappa}$ and let $T$ be a static set. Then $K^2x^2\le N^{1-2\kappa}N^{2\kappa}=N$, so $K<N$. Suppose some $t\in T$ has $t>K$. As $|T|=K$, some $v\in\{1,\dots,K\}$ is not in $T$. Then $\deg(v)\ge N/v-2\ge N/K-2$, and $\deg(t)\le\tau(t)+N/t\le Cx+N/(K+1)$. The static condition $\deg(v)\le\deg(t)$ gives $N/(K(K+1))\le Cx+2$. But $K(K+1)\le2K^2$ and $K^2x^2\le N$ give $N/(K(K+1))\ge x^2/2$, and $x^2/2-Cx-2=x(x/2-C)-2\ge2x-2>0$. This contradiction shows $T\subset\{1,\dots,K\}$, and $|T|=K$ gives $T=\{1,\dots,K\}$.
\end{proof}
\uses{Lemmas~\ref{lem:divisor}, \ref{lem:mono}, \ref{lem:sandwich}; Corollary~\ref{cor:profile}.}

\noindent\emph{Referee check.} The attack set $T$ must not depend on the orientation. Otherwise $\cD_\rho(N)-T$ is not an independently oriented copy of $G_N-T$, and the coupling argument breaks down. This is why in- and out-degree attacks are excluded.

\begin{remark}[Edge share of the hubs]\label{rem:edges}\status{PROVEN}\lean{card_hubEdges, card_edges_zero, abs_card_hubEdges_sub_le, abs_card_edges_sub_le, hub_edge_share}
An edge meets $\{1,\dots,K\}$ if and only if its smaller endpoint $d$ is at most $K$. So $\{1,\dots,K\}$ meets exactly $\sum_{d\le K}(\lfloor N/d\rfloor-1)=N\log K+O(N)$ of the $\sum_{d\le N}(\lfloor N/d\rfloor-1)=N\log N+O(N)$ edges, for $1\le K\le N$; both $O(N)$ terms are at most $2N$ in absolute value. Hence, if $\log(K+1)/\log N\to\theta$, the hubs meet a fraction $\theta+o(1)$ of the edges.
\end{remark}

\begin{remark}[Why a hub clique does not suffice for $\theta>1/3$]\label{rem:clique}\status{PROVEN; not used}\lean{remark_7_2, card_obst_ge, obst_mem, obst_goodSet, dvd_three_primes}
A natural first approach takes the hub set $H=(K,N^{1/2-\delta}]$. It is strongly connected w.h.p., because any two hubs have at least $N^{2\delta}$ common multiples. One then attaches every $m$ with many divisors in $H$. This captures only the integers having a divisor in $(K,N^{1/2-\delta}]$. For $\theta\le1/3$ these are asymptotically almost all of $F_1$ (compare the sub-sum lemma, Appendix~\ref{app:subsum}). (These two heuristic asides are not formalised.) For $1/3<\theta<1/2$, however, integers $m=p_1p_2p_3s\le N$ with primes $p_i\in(N^{1/4},N^{1/3}]$ and a cofactor $s\le N^{\gamma}$, $\gamma<\theta-\frac13$, lie in $F_1$ and have no divisor in $(K,\sqrt N\,]$, so none in $(K,N^{1/2-\delta}]$. Indeed, a divisor of $p_1p_2p_3s$ is either a multiple of some $p_ip_j$ ($i\ne j$), hence $>\sqrt N$, or a divisor of some $p_ks$, hence $\le N^{1/3+\gamma}\le K$ for large $N$. Their number is $\gg N$. To see this, take $\gamma=(\theta-\frac13)/3$ and put $\ell_j=\frac{1-\gamma}3+\frac{j\gamma}9$, so that $(N^{\ell_0},N^{\ell_3}]=(N^{(1-\gamma)/3},N^{1/3}]\subset(N^{1/4},N^{1/3}]$. Take $p_{j+1}$ prime in $(N^{\ell_j},N^{\ell_{j+1}}]$ for $j=0,1,2$. Then $p_1<p_2<p_3$ and $p_1p_2p_3>N^{1-\gamma}$, so every $s\le N/(p_1p_2p_3)$ satisfies $s<N^\gamma$, and $m=p_1p_2p_3s$ determines $(p_1,p_2,p_3,s)$. By Lemma~\ref{lem:mertens}(b), the number of these $m$ is, for large $N$,
\[
\sum_{p_1,p_2,p_3}\Bigl\lfloor\frac N{p_1p_2p_3}\Bigr\rfloor\ \ge\ \frac N2\prod_{j=0}^{2}\ \sum_{N^{\ell_j}<p\le N^{\ell_{j+1}}}\frac1p\ \ge\ \frac N2\prod_{j=0}^{2}\frac12\log\frac{\ell_{j+1}}{\ell_j}.
\] The method above avoids this obstruction. Such an $m$ lies in $S_M$ itself if $m\le M$. Otherwise, if the $p_i$ are distinct, its three divisors $m/p_i=p_jp_ks\in(K,M]$ lie in $S_M$, so $s(m)\ge3$, and $m$ attaches as in Lemma~\ref{lem:attach}; further divisors in $S_M$ come from the small prime factors of $s$, as in Lemma~\ref{lem:core}. The connectivity of the lower layer is supplied deterministically by Lemma~\ref{lem:doubling}, not by a clique.
\end{remark}

\begin{remark}[Relation to~\cite{KP}]\label{rem:KP}\status{PROVEN}\lean{kp_corollary_2}
The lower bound of~\cite{KP} uses the triangles $\{1,d,m\}$ and needs the vertex $1$. The present argument uses $2$-paths through common \emph{multiples} of divisibility pairs in the lower layer. It gives a new proof of~\cite[Corollary~2]{KP}, the case $K=0$ of Corollary~\ref{cor:profile}, and extends it to all $K$. The case $K=0$ is also checked formally (\texttt{kp\_corollary\_2}, and \texttt{Checks/ProfileCheck.lean}).
\end{remark}

\subsection{Open problems}\label{sec:open-problems}
\begin{enumerate}[label=(O\arabic*)]
\item \status{OPEN} \emph{True rate.} Determine the order of $\Psi(N,N/(K+1))-\E[\Phi_K]$. Theorem~\ref{thm:rate} gives $O(N/\log\log N)$. Vertices of degree $1$ in $F_1$ are never in an SCC with at least two vertices. This gives $N-\E[\Phi_0]\ge\pi(N)-\pi(N/2)\sim N/(2\log N)$ (the primes in $(N/2,N]$) and $\Psi(N,B)-\E[\Phi_1]\ge\pi(N/2)-\pi(N/3)\sim N/(6\log N)$ (the primes in $(N/3,N/2]$); the two asymptotics follow from the prime number theorem~\cite{Had,dlVP}. Is the error $N(\log N)^{-1+o(1)}$, uniformly in $K\le N^{1-\eta}$? The bottleneck in our argument is the variance bound (Lemma~\ref{lem:turan}) and the thin layer $Y<\Pp(m)\le B$.
\item \status{OPEN} \emph{Fluctuations.} Find the variance and the limiting distribution of $\Phi_K$; this is~\cite[Question~21]{KP} in the presence of hub removal. Corollary~\ref{cor:structure} gives concentration only at scale $o(N)$.
\item \status{OPEN} \emph{Orientation-dependent attacks.} Determine the limit profile when hubs are removed by in-degree or out-degree in the realised orientation, which is adaptive to the randomness.
\item \status{OPEN} \emph{Small $\rho$.} Remark~\ref{rem:rho} allows $\min(\rho,1-\rho)\gg1/\log\log N$. Where does the profile change when $\rho=\rho_N\to0$ faster?
\item \status{OPEN} \emph{Diameter.} What is the diameter of the giant SCC after hub removal? Compare~\cite[Conjecture~6]{KP}.
\end{enumerate}

\section{The sub-sum lemma}\label{app:subsum}
%======================================================================

The preliminary notes suggested proving the case $\theta\le1/3$ through the Poisson--Dirichlet description of the large prime factors together with the following deterministic lemma. Our proof of Theorem~\ref{thm:main} makes this route unnecessary, but we record the lemma with a proof.

\begin{lemma}[Sub-sum lemma]\label{lem:subsum}\status{PROVEN; elementary}\lean{subsum, subsum_of_mem_Icc, subsum_sharp}
Let $x_1,\dots,x_n$ be real numbers with $\sum_ix_i=1$, and let $0<\theta\le1/3$. If $x_i<1-\theta$ for every $i$, then there is a set $I\subset\{1,\dots,n\}$ with $\sum_{i\in I}x_i\in[\theta,\frac12]$. This fails for every $\theta\in(\frac13,\frac12]$: take $(x_1,x_2,x_3)=(\frac13,\frac13,\frac13)$.
\end{lemma}
In the setting of the preliminary notes, $x_1\ge x_2\ge\dots>0$, and the hypothesis is $x_1<1-\theta$. Neither monotonicity nor positivity is needed. The same proof works for infinite sequences of nonnegative terms, which case is not formalised.
\begin{proof}
We use that if $S=\sum_{i\in I}x_i\in[\theta,1-\theta]$, then either $S\le\frac12$, or $1-S=\sum_{i\notin I}x_i\in[\theta,\frac12)$.

If some $x_i\ge\theta$, take $I=\{i\}$; then $x_i\in[\theta,1-\theta)$.

Otherwise all $x_i<\theta$. The partial sums $S_k=x_1+\dots+x_k$ start at $S_0=0<\theta$ and end at $S_n=1\ge\theta$. Let $k$ be minimal with $S_k\ge\theta$. Then $S_{k-1}<\theta$, so $S_k=S_{k-1}+x_k<2\theta\le1-\theta$. So $S_k\in[\theta,1-\theta]$, and we take $I=\{1,\dots,k\}$ or its complement.

For the counterexample with $\theta\in(\frac13,\frac12]$: every $x_i=\frac13<1-\theta$, and the sub-sums are $0,\frac13,\frac23,1$, none of which lies in $[\theta,\frac12]$.
\end{proof}

\subsection*{The formal model}

\section{Proofs of Lemmas~\ref{lem:mertens} and~\ref{lem:dickman}}\label{app:classical}
%======================================================================

The proofs follow the formal proofs in the Lean files \texttt{Mertens}, \texttt{Mertens2}, \texttt{Dickman}, \texttt{Buchstab}, \texttt{PrimeSum} and \texttt{DickmanThm} of the directory \texttt{lean/HubRemoval/}. Throughout, $n\ge1$ is an integer, $\vartheta(n):=\sum_{p\le n}\log p$ and $S(n):=\sum_{p\le n}\frac{\log p}p$.

\subsection*{Proof of Lemma~\ref{lem:mertens}}
\emph{(i) Chebyshev.} $\vartheta(n)\le n\log4$, because $\prod_{p\le n}p\le4^n$ (the classical bound through the central binomial coefficient; it is in Mathlib).

\emph{(ii) $S(n)\le\log n+\log4$.} By Legendre's formula, $v_p(n!)\ge\lfloor n/p\rfloor\ge n/p-1$ for $p\le n$. Every prime factor of $n!$ is at most $n$, and $n!\le n^n$. Hence
\[
\sum_{p\le n}\Bigl(\frac np-1\Bigr)\log p\le\sum_{p\le n}v_p(n!)\log p=\log n!\le n\log n,
\]
so $nS(n)\le n\log n+\vartheta(n)\le n\log n+n\log4$.

\emph{(iii) $\sum_{p\le n}\frac{\log p}{p(p-1)}\le2$.} Put $h(k):=(\log k+2)/k$. For $k\ge2$,
\[
h(k-1)-h(k)=\frac{\log k-k\log\frac k{k-1}+2}{k(k-1)}\ge\frac{\log k}{k(k-1)},
\]
because $k\log\frac k{k-1}\le\frac k{k-1}\le2$. Summing over $2\le k\le n$ gives $\sum_{p\le n}\frac{\log p}{p(p-1)}\le h(1)=2$.

\emph{(iv) Part (a).} Since $\frac{\log p}{p-1}=\frac{\log p}p+\frac{\log p}{p(p-1)}$, (ii) and (iii) give $\sum_{p\le n}\frac{\log p}{p-1}\le\log n+\log4+2=\log n+c_0$. For real $y\ge2$ apply this to $n=\lfloor y\rfloor$.

\emph{(v) Mertens' first theorem: $|S(n)-\log n|\le3$.} The upper bound is (ii), as $\log4<3$. For the lower bound, Legendre's formula gives $(p-1)v_p(n!)\le n$. So $\log n!\le n\sum_{p\le n}\frac{\log p}{p-1}\le n(S(n)+2)$ by (iii). Also $\log n!\ge n\log n-n$, because $n^n/n!\le e^n$. Hence $S(n)\ge\log n-3$.

\emph{(vi) Part (b) for integers $2\le y\le w$.} Since $S(k)-S(k-1)=\frac{\log k}k$ if $k$ is prime and $0$ otherwise,
\[
\sum_{y<p\le w}\frac1p=\sum_{y<k\le w}\frac{S(k)-S(k-1)}{\log k}.
\]
Write $S(k)=\log k+E(k)$, so $|E(k)|\le3$ by (v). Put $f(k):=1/\log k$, which is positive and non-increasing for $k\ge2$. By Abel summation,
\[
\sum_{y<k\le w}\bigl(E(k)-E(k-1)\bigr)f(k)=E(w)f(w)-E(y)f(y)+\sum_{y<k\le w}E(k-1)\bigl(f(k-1)-f(k)\bigr),
\]
whose absolute value is at most $3f(w)+3f(y)+3(f(y)-f(w))=6/\log y$. The remaining part is $\sum_{y<k\le w}(v_k-u_k)/v_k$ with $u_k:=\log(k-1)$ and $v_k:=\log k$. Since $\frac{v-u}v\le\log\frac vu\le\frac{v-u}u$, $v_k-u_k=\log\frac k{k-1}\le\frac1{k-1}$ and $u_k,v_k\ge\log y$,
\[
0\le\log\frac{v_k}{u_k}-\frac{v_k-u_k}{v_k}\le\frac{(v_k-u_k)^2}{u_kv_k}\le\frac1{(k-1)^2\log^2y}.
\]
As $\sum_{y<k\le w}\log\frac{v_k}{u_k}=\log\log w-\log\log y$ and $\sum_{j\ge2}j^{-2}\le1$, we get
\[
\Bigl|\sum_{y<p\le w}\frac1p-\log\frac{\log w}{\log y}\Bigr|\le\frac6{\log y}+\frac1{\log^2y}\le\frac8{\log y},
\]
using $\log y\ge\log2>\frac12$.

\emph{(vii) Part (b) for reals.} For real $x\ge2$ and $n=\lfloor x\rfloor$ we have $\log x-\log n\le\log\frac{n+1}n\le\frac1n$, so
\[
0\le\log\log x-\log\log n\le\frac{\log x-\log n}{\log n}\le\frac1{n\log n}\le\frac1{2\log n},
\]
and $\frac1{\log n}\le\frac2{\log x}$, since $x<n+1\le n^2$. Let $Y=\lfloor y\rfloor$ and $W=\lfloor w\rfloor$. The primes in $(y,w]$ are those in $(Y,W]$. By (vi), the error is at most $\frac8{\log Y}+\frac1{2\log W}+\frac1{2\log Y}\le\frac9{\log Y}\le\frac{18}{\log y}$.

\emph{(viii) Part (c).} Let $Z=\lfloor z\rfloor\ge2$. By (vi) with $y=2$ and $w=Z$, $\sum_{p\le Z}\frac1p\ge\sum_{2<p\le Z}\frac1p\ge\log\log Z-\log\log2-\frac8{\log2}\ge\log\log Z-12$, as $\log\log2<0$ and $8/\log2<12$. By (vii), $\log\log z\le\log\log Z+\frac1{2\log Z}\le\log\log Z+1$. Hence $\sum_{p\le z}\frac1p\ge\log\log z-13$.

\emph{(ix) Part (d).} At most $\sqrt x$ primes are $\le\sqrt x$. Each prime $p\in(\sqrt x,x]$ has $\log p>\frac12\log x$, so there are at most $\vartheta(x)/(\frac12\log x)\le2x\log4/\log x$ of them, by (i). Also $\sqrt x\log x=2\sqrt x\log\sqrt x\le2\sqrt x(\sqrt x-1)\le2x$, so $\sqrt x\le2x/\log x$. Hence $\pi(x)\le(2+2\log4)x/\log x\le5x/\log x$, as $\log4<1.39$. \qed

\subsection*{Proof of Lemma~\ref{lem:dickman}: the function}
\emph{(i) Construction.} Let $\rho_0\equiv1$ on $\mathbb R$, and for $n\ge0$ let
\[
\rho_{n+1}(u):=\begin{cases}\rho_n(u),&u\le n+1,\\ \rho_n(n+1)-\int_{n+1}^u\rho_n(w-1)\,\frac{dw}w,&u\ge n+1.\end{cases}
\]
By induction each $\rho_n$ is continuous, and $\rho_m=\rho_n$ on $(-\infty,m+1]$ for $m\le n$. Put $\dick(u):=\rho_n(u)$ for any $n$ with $u\le n+1$. Then $\dick$ is continuous, $\dick=1$ on $(-\infty,1]$, and for $n+1\le u\le n+2$ we have $\dick(u)=\dick(n+1)-\int_{n+1}^u\dick(w-1)\,\frac{dw}w$, since $w-1\le n+1$ there.

\emph{(ii) The integral equation.} Adding the relations of (i) over $[1,2],[2,3],\dots$ gives $\dick(u)=1-\int_1^u\dick(w-1)\frac{dw}w$ for $u\ge1$, and $\dick(u)=\dick(v)-\int_v^u\dick(w-1)\frac{dw}w$ for $1\le v\le u$. By the fundamental theorem of calculus, $\dick$ is differentiable at every $u>1$ with $\dick'(u)=-\dick(u-1)/u$. On $[1,2]$ the integrand is $1/w$, so $\dick(u)=1-\log u$.

\emph{(iii) The integrated form.} $G(v):=v\dick(v)-\int_{v-1}^v\dick$ is continuous, and for $v>1$, $G'(v)=\dick(v)+v\dick'(v)-(\dick(v)-\dick(v-1))=0$. By the mean value theorem, $G(u)=G(1)=1-\int_0^11=0$ for $u\ge1$.

\emph{(iv) Positivity.} Suppose $\dick(u)\le0$ for some $u$. The set $\{v:\dick(v)\le0\}$ is closed, non-empty and bounded below by $1$; let $u_0$ be its least element. Then $u_0>1$ and $\dick>0$ on $(-\infty,u_0)$. By (iii), $u_0\dick(u_0)=\int_{u_0-1}^{u_0}\dick>0$, which contradicts $\dick(u_0)\le0<u_0$.

\emph{(v) Monotonicity.} For $u>1$, $\dick'(u)=-\dick(u-1)/u<0$, so $\dick$ is strictly decreasing on $[1,\infty)$. Hence $\dick$ is non-increasing on $\mathbb R$ and $\dick\le\dick(1)=1$.

\emph{(vi) Uniqueness.} Let $g$ be continuous on $[0,\infty)$, with $g=1$ on $[0,1]$ and $g'(u)=-g(u-1)/u$ for $u>1$. We show $g=\dick$ on $[0,n+1]$ by induction on $n$; the case $n=0$ is given. If $g=\dick$ on $[0,n+1]$, then on $(n+1,n+2)$ the derivative of $g-\dick$ is $-(g(u-1)-\dick(u-1))/u=0$. By the mean value theorem and continuity, $g-\dick$ is constant on $[n+1,n+2]$, and it vanishes at $n+1$. \qed

\subsection*{Buchstab's identity and a prime-sum estimate}

\begin{lemma}[Buchstab's identity]\label{lem:buchstab}\status{PROVEN}\lean{buchstab, buchstab_real, psi_prime_step, psi_nonprime_step, psi_eq_self_of_le}
For real $x$ and $0\le y\le z$,
\[
\Psi(x,z)=\Psi(x,y)+\sum_{y<p\le z}\Psi(x/p,p).
\]
Moreover $\Psi(X,Y)=X$ for integers $0\le X\le Y$.
\end{lemma}
\begin{proof}
First let $X=\lfloor x\rfloor$ and let $Y\le Z$ be integers; we induct on $Z\ge Y$. If $Z+1$ is not prime, the $(Z+1)$-smooth numbers are the $Z$-smooth ones. If $p=Z+1$ is prime, the $n\le X$ that are $p$-smooth but not $(p-1)$-smooth are exactly the $n=pm$ with $m\le\lfloor X/p\rfloor$ and $m$ $p$-smooth, and $m\mapsto pm$ is injective. So $\Psi(X,p)=\Psi(X,p-1)+\Psi(\lfloor X/p\rfloor,p)$. For real arguments use $\lfloor x/p\rfloor=\lfloor\lfloor x\rfloor/p\rfloor$ and $y<p\iff\lfloor y\rfloor<p$. Finally, if $X\le Y$, every $n\in[1,X]$ has all prime factors $\le n\le Y$.
\end{proof}

\begin{lemma}[Prime sums as Riemann sums]\label{lem:primesum}\status{PROVEN}\lean{primeSum_integral, step_bound, sum_primesBetween_inv, log_div_log_mem}
Let $0<a\le b$, and let $g$ be continuous and non-increasing on $[a,b]$ with $0\le g\le G$ there. For $x>1$ and a prime $p$ put $w_p:=\log x/\log p$. Then for every $\varepsilon>0$ there is $x_0$ such that for all $x\ge x_0$ and all $u\in[a,b]$,
\[
\Bigl|\sum_{x^{1/u}<p\le x^{1/a}}\frac{g(w_p)}p-\int_a^u g(w)\,\frac{dw}w\Bigr|\le\varepsilon .
\]
\end{lemma}
\begin{proof}
For $a\le s\le t\le b$ let $P(s,t)$ be the set of primes $p$ with $x^{1/t}<p\le x^{1/s}$; for these, $s\le w_p<t$. Assume $x^{1/b}\ge2$. Lemma~\ref{lem:mertens}(b) with $y=x^{1/t}$ and $w=x^{1/s}$, where $\log(\log w/\log y)=\log(t/s)$ and $C_1/\log y=18t/\log x$, gives $|A-\log(t/s)|\le\eta:=18b/\log x$ for $A:=\sum_{p\in P(s,t)}\frac1p$. As $g$ is non-increasing, $\sum_{p\in P(s,t)}g(w_p)/p$ lies in $[g(t)A,g(s)A]$, and $\int_s^tg(w)\frac{dw}w$ lies in $[g(t)\ell,g(s)\ell]$ with $\ell:=\log(t/s)\in[0,(t-s)/a]$. Hence the two differ by at most
\[
\bigl(g(s)-g(t)\bigr)\frac{t-s}a+G\eta .
\]
Fix an integer $m\ge1$ and put $t_i:=a+i(u-a)/m$ for $0\le i\le m$. The sets $P(t_i,t_{i+1})$ partition $P(a,u)$, and the integrals over $[t_i,t_{i+1}]$ add up to the integral over $[a,u]$. Summing the step bounds, the first terms telescope, and $t_{i+1}-t_i\le(b-a)/m$, so the difference in the lemma is at most
\[
\bigl(g(a)-g(u)\bigr)\frac{b-a}{ma}+mG\eta\le\frac{G(b-a)}{ma}+\frac{18bmG}{\log x}.
\]
Choose $m>2G(b-a)/(a\varepsilon)$, so the first term is $\le\varepsilon/2$. The second term is $\le\varepsilon/2$ once $\log x\ge36bmG/\varepsilon$, and $x^{1/b}\ge2$ for large $x$. None of these conditions depends on $u$.
\end{proof}

\subsection*{Proof of Lemma~\ref{lem:dickman}: the asymptotic}
Write $\varphi(x,u):=\Psi(x,x^{1/u})/x$. We show by induction on $n\ge2$: for every $\varepsilon>0$, for all large $x$, $|\varphi(x,u)-\dick(u)|\le\varepsilon$ for all $u\in[1,n]$. For $U\ge1$ this gives the lemma with $n=\lceil U\rceil+2$. \lean{psi_dickman_base, psi_dickman_step, psi_dickman_nat}

\emph{Base case $n=2$.} Let $x\ge4$ and $u\in[1,2]$, so $x^{1/u}\ge\sqrt x\ge2$. Lemma~\ref{lem:buchstab} with $y=x^{1/u}$ and $z=x$ gives $\Psi(x,x^{1/u})=\lfloor x\rfloor-\sum_{x^{1/u}<p\le x}\Psi(x/p,p)$. For these $p$ we have $p>\sqrt x$, so $x/p<p$ and $\Psi(x/p,p)=\lfloor x/p\rfloor\in(x/p-1,x/p]$. With $A:=\sum_{x^{1/u}<p\le x}\frac1p$ and Lemma~\ref{lem:mertens}(b),(d),
\[
\bigl|\Psi(x,x^{1/u})-x(1-\log u)\bigr|\le1+\pi(x)+x|A-\log u|\le1+\frac{5x}{\log x}+\frac{36x}{\log x},
\]
since $C_1/\log x^{1/u}=18u/\log x\le36/\log x$. As $\dick(u)=1-\log u$ on $[1,2]$, we get $|\varphi(x,u)-\dick(u)|\le1/x+41/\log x\le\varepsilon$ for $x\ge\max(4,2/\varepsilon)$ and $\log x\ge82/\varepsilon$.

\emph{Step from $n$ to $n+1$, $n\ge2$.} Let $\varepsilon>0$. By the induction hypothesis there is $T$ with $|\varphi(t,v)-\dick(v)|\le\varepsilon/4$ for all $t\ge T$ and $v\in[1,n]$. Apply Lemma~\ref{lem:primesum} with $a=n$, $b=n+1$, $g(w)=\dick(w-1)$ (continuous, non-increasing, $0<g\le1=G$) and $\varepsilon/4$. Let $x$ be so large that this applies, $\sqrt x\ge\max(T,1)$, $x^{1/(n+1)}\ge2$ and $\log x\ge18(n+1)$. For $u\in[1,n]$ the induction hypothesis applies, so let $u\in[n,n+1]$. Lemma~\ref{lem:buchstab} with $y=x^{1/u}$ and $z=x^{1/n}$ gives
\[
\varphi(x,u)=\varphi(x,n)-\sum_{p\in P}\frac{\Psi(x/p,p)}x,\qquad P:=\{p:\ x^{1/u}<p\le x^{1/n}\}.
\]
For $p\in P$ we have $w_p=\log x/\log p\in[n,u)$, so $v_p:=w_p-1\in[n-1,n)\subset[1,n]$, and $(x/p)^{1/v_p}=\exp\bigl((\log x-\log p)/v_p\bigr)=p$. Also $p\le x^{1/n}\le\sqrt x$, so $x/p\ge\sqrt x\ge T$. The induction hypothesis at $t=x/p$ and $v=v_p$ gives $|\Psi(x/p,p)/(x/p)-\dick(w_p-1)|\le\varepsilon/4$, that is, $|\Psi(x/p,p)/x-\dick(w_p-1)/p|\le\varepsilon/(4p)$. By Lemma~\ref{lem:mertens}(b), $\sum_{p\in P}\frac1p\le\log\frac un+\frac{18u}{\log x}\le1+1$. Hence
\[
\Bigl|\sum_{p\in P}\frac{\Psi(x/p,p)}x-\sum_{p\in P}\frac{\dick(w_p-1)}p\Bigr|\le\frac\varepsilon2 .
\]
By Lemma~\ref{lem:primesum}, $\bigl|\sum_{p\in P}\dick(w_p-1)/p-\int_n^u\dick(w-1)\frac{dw}w\bigr|\le\varepsilon/4$. By (ii) of the construction above, $\dick(u)=\dick(n)-\int_n^u\dick(w-1)\frac{dw}w$. Finally $|\varphi(x,n)-\dick(n)|\le\varepsilon/4$ by the induction hypothesis at $t=x\ge T$ and $v=n$. Together, $|\varphi(x,u)-\dick(u)|\le\frac\varepsilon4+\frac\varepsilon2+\frac\varepsilon4=\varepsilon$. \qed
